# Source: OpenAI API Platform audit logs (GET /v1/organization/audit_logs), one object from
# data[] per HTTP Ingest record. The API is pull only, so a bridge reads it and sends it.
# Standalone parser for the Authentication schema: login.succeeded, login.failed,
# logout.succeeded and logout.failed in the API Platform organization.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

json = JSON(originalData$)
sensorType$ = "OpenAI API Platform"
sensorEventId$ = json["$.id"]
# effective_at is Unix seconds; the field takes microseconds.
eventTimeUsec$ = INT(STRING(json["$.effective_at"]), 10) * 1000000

t = json["$.type"]
targetUserName$ = json["$.actor.session.user.email"]
sourceAddress$ = json["$.actor.session.ip_address"]
userAgent$ = json["$.actor.session.user_agent"]

isLogin = CONTAINS(t, "login.")
failed = CONTAINS(t, ".failed")
action$ = IF isLogin THEN "LOGON" ELSE "LOGOFF"
action$ = IF failed THEN "FAILURE" ELSE action$
authCategory$ = IF isLogin THEN "ACCOUNT_LOGON" ELSE "ACCOUNT_LOGOFF"
actionResult$ = IF failed THEN "FAILED" ELSE "SUCCESS"
# The details object is keyed by the event type, for example "login.failed": {...}.
# OpenAI doesn't list its error codes, so a failure is OTHER_ERROR with the message kept.
failureCategory$ = IF failed THEN "OTHER_ERROR" ELSE NULL
extraFailurereason$ = json["$['" + t + "'].error_message"]

# Keys with no Auth field of their own go to event_metadata.
typ.key$ = "openai_event_type"
typ.value$ = t
uid.key$ = "openai_user_id"
uid.value$ = json["$.actor.session.user.id"]
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [typ, uid]

# Optional keys, added only when OpenAI sends them.
code.key$ = "openai_error_code"
code.value$ = json["$['" + t + "'].error_code"]
eventMetadata$.record$ = IF code.value$ THEN eventMetadata$.record$ + [code] ELSE eventMetadata$.record$
country.key$ = "source_country"
country.value$ = json["$.actor.session.ip_address_details.country"]
eventMetadata$.record$ = IF country.value$ THEN eventMetadata$.record$ + [country] ELSE eventMetadata$.record$
asn.key$ = "source_asn"
asn.value$ = json["$.actor.session.ip_address_details.asn"]
eventMetadata$.record$ = IF asn.value$ THEN eventMetadata$.record$ + [asn] ELSE eventMetadata$.record$
ja3.key$ = "tls_ja3"

eventMetadata$.record$ = IF ja3.value$ THEN eventMetadata$.record$ + [ja3] ELSE eventMetadata$.record$
