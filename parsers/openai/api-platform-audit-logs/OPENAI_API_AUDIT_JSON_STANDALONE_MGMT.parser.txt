# Source: OpenAI API Platform audit logs (GET /v1/organization/audit_logs), one object from
# data[] per HTTP Ingest record. The API is pull only, so a bridge reads it and sends it.
# Standalone parser for the ManagementEvent schema: every event type in OpenAI's list except
# login and logout (API keys, projects, users, invites, roles, groups, service accounts,
# rate limits, IP allowlists, SCIM, certificates, organization and tenant settings).
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

json = JSON(originalData$)
sensorType$ = "OpenAI API Platform"
sensorEventId$ = json["$.id"]
# effective_at is Unix seconds; the field takes microseconds.
eventTimeUsec$ = INT(STRING(json["$.effective_at"]), 10) * 1000000

t = json["$.type"]
operationName$ = t
# The project the action was scoped to. Admin API key actions use the default project.
namespace$ = json["$.project.id"]

# The details object is keyed by the event type, for example "api_key.created": {"id": ...}.
# operation$ is the schema's free-text summary: the event type and the ID of what it changed.
target = json["$['" + t + "'].id"]
operation$ = IF target THEN t + " " + target ELSE t

# The actor is a signed-in user (session) or an API key that belongs to a user or a
# service account.
username$ = json["$.actor.session.user.email"]
username$ = IF username$ THEN username$ ELSE json["$.actor.api_key.user.email"]
serviceAccount = json["$.actor.api_key.service_account.id"]
username$ = IF serviceAccount THEN "service_account " + serviceAccount ELSE username$

# Every action arrives over the web or the API, so it is remote. Only sessions carry an IP.
# The schema has no address field; host_id takes an IP address, so the client IP goes there.
remoteOperation$ = TRUE
hostId$ = json["$.actor.session.ip_address"]
