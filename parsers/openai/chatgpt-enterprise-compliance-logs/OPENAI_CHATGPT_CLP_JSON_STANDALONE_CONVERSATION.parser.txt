# object per HTTP Ingest record (the API is pull only, so a bridge downloads the files).
# Standalone parser for the CloudAudit schema: who sent or received a message, when, with
# which model, client, GPT or Project. Each record is one message snapshot.
# Privacy: these records carry prompt and response text. This parser maps metadata only.
# Message text, quotes, custom instructions, conversation titles, file names and GPT or
# Project names never go into a normalized field. original_data keeps whatever the
# bridge sent, so redact content before it leaves the bridge.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

json = JSON(originalData$)
sensorType$ = "ChatGPT Enterprise"
sensorEventId$ = json["$.event_id"]
# ISO 8601 in UTC with up to microseconds.
eventTimeUsec$ = DATETIME(json["$.timestamp"], "2006-01-02T15:04:05.999999999Z07:00")

eventName$ = json["$.type"]
eventSource$ = "ChatGPT"
# user for a prompt, assistant for a model response.
eventType$ = json["$.message.author.type"]
# The actor is the workspace member whose conversation it is, for both prompts and responses.
userName$ = json["$.actor.user_email"]
userId$ = json["$.actor.user_id"]
userType$ = json["$.actor.type"]
recipientAccountId$ = json["$.principal.id"]

# Resources: the conversation and the message, plus the GPT and Project when set.
conv.resourceId$ = json["$.conversation.id"]
conv.resourceType$ = "conversation"
conv.resourceAccountId$ = json["$.principal.id"]
msg.resourceId$ = json["$.message.id"]
msg.resourceType$ = "message"
msg.resourceAccountId$ = json["$.principal.id"]
resources$ = [conv, msg]
gpt.resourceId$ = json["$.conversation.gpt_id"]
gpt.resourceType$ = "gpt"
gpt.resourceAccountId$ = json["$.principal.id"]
