# HTTP Ingest record (the API is pull only, so a bridge downloads the files and sends them).
# Standalone parser for the ManagementEvent schema: workspace settings, SSO and SCIM, roles,
# groups, invites, keys, sharing and Compliance API calls, by admins and by users.
# action_data varies by action and can hold free text (custom and system instructions), so
# it is not mapped; it stays in original_data.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

json = JSON(originalData$)
sensorType$ = "ChatGPT Enterprise"
sensorEventId$ = json["$.event_id"]
# ISO 8601 in UTC with up to microseconds.
eventTimeUsec$ = DATETIME(json["$.timestamp"], "2006-01-02T15:04:05.999999999Z07:00")

operationName$ = json["$.action"]
# SUCCESS, BLOCKED or ERROR.
result$ = json["$.action_result"]
# ADMIN or STANDARD_USER. The schema has no privilege field, and operation$ is its free-text
# summary of the operation.
operation$ = json["$.action_privilege"]
channel$ = json["$.type"]
# The ChatGPT workspace (or API Platform organization) the action belongs to.
namespace$ = json["$.principal.id"]

# The actor is a user, a Compliance API key, or someone in Slack or Teams using a workspace agent.
actorType = json["$.actor.type"]
