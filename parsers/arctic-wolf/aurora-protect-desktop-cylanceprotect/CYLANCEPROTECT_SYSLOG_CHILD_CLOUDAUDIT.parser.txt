# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Aurora Protect Desktop (CylancePROTECT) audit log over syslog: what console users and
# API applications did in the Aurora Endpoint Security console, such as policy, zone, user, global
# list and device changes. Sign-ins go to the Auth child.
# Child of CYLANCEPROTECT_SYSLOG_PARENT: each audit event to CloudAudit, with the event name (such
# as PolicyEdit or ThreatGlobalQuarantine) as the event name and the console user as the user.
# evType, evName, eventType$, sensorType$ and eventTimeUsec$ come from the parent.

eventSource$ = "Aurora Endpoint Security"
eventName$ = evName
managementEvent$ = TRUE
lname = LOWERCASE(evName)
readOnly$ = CONTAINS(lname, "download") OR CONTAINS(lname, "show") OR lname == "focusdataadd"

# Message runs to the last ", User: ", which names the console user: a name and the email address
# in parentheses, or the address alone. An Eco Id can follow the address inside the parentheses.
# Changes made through the API have no console user.
msg = originalData$|, Message: (?P<v>.*), User:
u = originalData$|.*, User: (?P<name>[^(,]*?)\s*\(\s*(?P<email>[^,)\s]+)
userName$ = IF u["email"] != "" THEN u["email"] ELSE NULL
userType$ = IF u["email"] != "" THEN "User" ELSE NULL
# User changes name the user they changed in the message, as "User: <name>".
tgt = msg["v"]|^User: (?P<v>.+)$
targetUserName$ = IF (lname == "useradd" OR lname == "useredit" OR lname == "userremove") AND tgt["v"] != "" THEN tgt["v"] ELSE NULL
src = originalData$|, Source IP: (?P<v>[0-9A-Fa-f.:]+)
sourceAddress$ = IF src["v"] != "" THEN src["v"] ELSE NULL

d1.key$ = "message"
d1.value$ = msg["v"]
# Assign the list first: reading additionalEventData$.record$ before it is set fails the event.
additionalEventData$.record$ = []
additionalEventData$.record$ = IF msg["v"] != "" THEN additionalEventData$.record$ + [d1] ELSE additionalEventData$.record$
eco = originalData$|, Eco Id: (?P<v>[^,)\s]+)
d2.key$ = "eco_id"
d2.value$ = eco["v"]
additionalEventData$.record$ = IF eco["v"] != "" THEN additionalEventData$.record$ + [d2] ELSE additionalEventData$.record$
