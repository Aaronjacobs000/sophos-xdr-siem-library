!SAMPLE=Oct  7 23:02:11 gw01 dnsmasq[812]: query[A] update.example.com from 192.0.2.50
!CONFIRMWITH=PATTERN
!CONFIRMSTRING=dnsmasq(\[\d+\])?:\s+query\[[A-Za-z0-9=]+\]\s+\S+\s+from\s
!SCHEMA=scwx.dnsquery

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-ngsiem-parser-studio
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: dnsmasq with --log-queries, plain text over syslog to the data collector.
# Standalone parser for the DNS schema. Only query lines match; forwarded, reply and cached
# lines stay in the generic schema. Format from log_query() in dnsmasq's cache.c:
#   query[<type>] <name> from <client>
# Setting query_name lets the Domain Generation Algorithms and Domain Watchlist detectors use these events.

m = originalData$|dnsmasq(?:\[\d+\])?:\s+query\[(?P<type>[A-Za-z0-9=]+)\]\s+(?P<name>\S+)\s+from\s+(?P<client>\S+)

# The syslog time has no year and no zone: DATETIME picks the year (TRUE) and reads UTC.
hdr = originalData$|^(?:<\d+>)?(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?P<host>\S+)\s
eventTimeUsec$ = DATETIME(hdr["time"], "Jan _2 15:04:05", TRUE)
hostId$ = hdr["host"]

sensorType$ = "dnsmasq"
queryName$ = m["name"]
sourceAddress$ = m["client"]

# query_type takes the DNS type number. dnsmasq logs the name, or type=<n> for types it can't name.
qtype = m["type"]
typeNumber = qtype|^type=(\d+)$
queryType$ = IF typeNumber != NULL THEN INT(typeNumber[1], 10) ELSE NULL
queryType$ = IF qtype == "A" THEN 1 ELSE queryType$
queryType$ = IF qtype == "NS" THEN 2 ELSE queryType$
queryType$ = IF qtype == "CNAME" THEN 5 ELSE queryType$
queryType$ = IF qtype == "SOA" THEN 6 ELSE queryType$
queryType$ = IF qtype == "PTR" THEN 12 ELSE queryType$
queryType$ = IF qtype == "MX" THEN 15 ELSE queryType$
queryType$ = IF qtype == "TXT" THEN 16 ELSE queryType$
queryType$ = IF qtype == "AAAA" THEN 28 ELSE queryType$
queryType$ = IF qtype == "SRV" THEN 33 ELSE queryType$
queryType$ = IF qtype == "HTTPS" THEN 65 ELSE queryType$
queryType$ = IF qtype == "ANY" THEN 255 ELSE queryType$
