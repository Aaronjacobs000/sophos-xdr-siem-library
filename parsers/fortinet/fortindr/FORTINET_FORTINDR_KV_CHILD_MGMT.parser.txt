username$ = IF user != "" AND user != "none" THEN user ELSE NULL
# ui is the interface and, when there is one, the address: GUI(192.0.2.10), ssh(192.0.2.11), init.

channel$ = IF uim["via"] != "" AND uim["via"] != "none" THEN uim["via"] ELSE NULL
clientHostname$ = IF uim["ip"] != "" THEN uim["ip"] ELSE NULL
# A settings change names the setting and the configuration path: changed settings of '<attr>'
# for '<path>'.

operationName$ = IF cs["path"] != "" THEN "changed settings" ELSE "user event"
namespace$ = IF cs["path"] != "" THEN cs["path"] ELSE NULL
