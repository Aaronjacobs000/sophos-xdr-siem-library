# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-ngsiem-parser-studio
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: FortiNDR logs in the default syslog format.
# Child of FORTINET_FORTINDR_KV_PARENT: administrator logins and logouts (user events with
# FortiNDR's login, login failed, login blocked, logout and time out messages) to Auth.
# kv, msg, logId, eventType$, sensorType$, sensorEventId$, hostId$ and eventTimeUsec$ come from
# the parent.

authSystem$ = "FortiNDR"
m = msg|^User (?P<u>\S+) (?P<what>login successfully|login failed|login blocked|logout|time out)\s+(?:from|on)\s+(?:(?P<via>[A-Za-z]+)\()?(?P<ip>[0-9A-Fa-f.:]*[0-9A-Fa-f])\)?
user = STRING(kv["user"])
user = IF user == "" OR user == "none" THEN m["u"] ELSE user
targetUserName$ = IF user != "" THEN user ELSE NULL
sourceAddress$ = IF m["ip"] != "" THEN m["ip"] ELSE NULL
logonApplicationFamily$ = IF m["via"] != "" THEN m["via"] ELSE NULL

what = m["what"]
loggedOff = what == "logout" OR what == "time out"
failed = what == "login failed" OR what == "login blocked"
authCategory$ = IF loggedOff THEN "ACCOUNT_LOGOFF" ELSE "ACCOUNT_LOGON"
action$ = "LOGON"
action$ = IF loggedOff THEN "LOGOFF" ELSE action$
action$ = IF failed THEN "FAILURE" ELSE action$
actionResult$ = IF failed THEN "FAILED" ELSE "SUCCESS"
# A blocked login names its reason, such as try_too_many_times.
blocked = msg|\sdue to (?P<r>\S+)
failureCategory$ = IF what == "login blocked" THEN "ACCOUNT_LOCKED_OUT" ELSE NULL
targetHostName$ = IF STRING(kv["devhost"]) != "" THEN STRING(kv["devhost"]) ELSE NULL

m1.key$ = "log_id"
m1.value$ = logId
m2.key$ = "msg"
m2.value$ = msg
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [m1, m2]
m3.key$ = "block_reason"
m3.value$ = blocked["r"]
eventMetadata$.record$ = IF blocked["r"] != "" THEN eventMetadata$.record$ + [m3] ELSE eventMetadata$.record$
