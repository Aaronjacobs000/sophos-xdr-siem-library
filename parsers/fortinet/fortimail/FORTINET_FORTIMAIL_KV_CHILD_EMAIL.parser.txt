dir = LOWERCASE(STRING(kv["direction"]))
direction$ = NULL
direction$ = IF dir == "in" THEN "INBOUND" ELSE direction$
direction$ = IF dir == "out" THEN "OUTBOUND" ELSE direction$

# disposition is English text, or a hex bitmask of actions: 0x01 Accept, 0x04 Reject, 0x20
# Quarantine, 0x80 Block, 0x4000 Quarantine to Review, 0x100000000 Domain Quarantine (and more).
# The bits are read from the hex digits.
disp = STRING(kv["disposition"])
dl = LOWERCASE(disp)
hex = disp|^0[xX](?P<h>[0-9A-Fa-f]+)$
h = hex["h"]
d1 = h|(?P<d>[0-9A-Fa-f])$
d2 = h|(?P<d>[0-9A-Fa-f])[0-9A-Fa-f]$
d4 = h|(?P<d>[0-9A-Fa-f])[0-9A-Fa-f]{3}$
d9 = h|(?P<d>[0-9A-Fa-f])[0-9A-Fa-f]{8}$
acceptBit = d1["d"] != "" AND CONTAINS("13579BDFbdf", d1["d"])
rejectBit = d1["d"] != "" AND CONTAINS("4567CDEFcdef", d1["d"])
blockBit = d2["d"] != "" AND CONTAINS("89ABCDEFabcdef", d2["d"])
quarBit = d2["d"] != "" AND CONTAINS("2367ABEFabef", d2["d"])
quarBit = quarBit OR (d4["d"] != "" AND CONTAINS("4567CDEFcdef", d4["d"])) OR (d9["d"] != "" AND CONTAINS("13579BDFbdf", d9["d"]))
blocked = rejectBit OR blockBit OR CONTAINS(dl, "reject") OR CONTAINS(dl, "block") OR CONTAINS(dl, "discard")
quarantined = quarBit OR CONTAINS(dl, "quarantine")
accepted = acceptBit OR CONTAINS(dl, "accept")
status$ = NULL
status$ = IF history AND accepted THEN "DELIVERED" ELSE status$
status$ = IF history AND quarantined THEN "QUARANTINED" ELSE status$
status$ = IF history AND blocked THEN "BLOCKED" ELSE status$

cls = STRING(kv["classifier"])
clsHex = cls|^0[xX](?P<k>[0-9A-Fa-f]{2})$
className = IF clsHex["k"] != "" THEN STRING(classNames["$." + UPPERCASE(clsHex["k"])]) ELSE cls
className = IF className == "" THEN cls ELSE className
quarantineReason$ = IF status$ == "QUARANTINED" AND className != "" THEN className ELSE NULL

# A blocked or quarantined message, or an antispam log, gets one threat record.
c = LOWERCASE(className)
virusName = STRING(kv["virus"])
ttype = "policy"
ttype = IF CONTAINS(c, "spam") OR CONTAINS(c, "rbl") OR CONTAINS(c, "bayesian") OR CONTAINS(c, "heuristic") OR CONTAINS(c, "banned word") OR CONTAINS(c, "dictionary") OR CONTAINS(c, "reputation") OR CONTAINS(c, "newsletter") OR CONTAINS(c, "behavior") THEN "spam" ELSE ttype
ttype = IF CONTAINS(c, "phishing") OR CONTAINS(c, "impersonation") THEN "phishing" ELSE ttype
ttype = IF CONTAINS(c, "virus") OR CONTAINS(c, "malware") OR CONTAINS(c, "sandbox") OR virusName != "" THEN "virus" ELSE ttype
ttype = IF history THEN ttype ELSE "spam"
t_main.name$ = IF virusName != "" THEN virusName ELSE className
