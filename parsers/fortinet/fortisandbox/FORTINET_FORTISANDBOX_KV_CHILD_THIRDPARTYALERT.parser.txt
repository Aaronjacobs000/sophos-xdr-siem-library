# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: FortiSandbox logs sent to a remote log server of type syslog, key=value:
#   <PRI><time> devhost=<name> devid=<serial> tzone=<seconds> tz=<zone> date=<date> time=<time>
#   logid=<id> type=<type> subtype=<subtype> level=<level> <fields>
# Child of the global parser Fortinet_Master_non-cef, which claims these lines and stores nothing
# from them. In Sophos Fusion, turn on Set Global Parent in Parser Script. The parser reads the
# whole line itself, not the parent's variables.
# Alert logs to the third-party alert schema: malware (a file the sandbox rated), and the network
# alerts netattack, netbotnet and neturl. Fortinet's log reference names the alert type both alert
# and malware or netalert, so the parser takes any of them.

kvm = originalData$|(?P<kv>(?:devhost|devid)=.*)$
# CEF() reads the pairs: it keeps a quoted value whole even when it holds " key=" text, which
# SPLIT_NAME_VALUES cuts in Sophos NG-SIEM, and it ends an unquoted value at the next " key=".
kv = CEF("CEF:0|Fortinet|FortiSandbox|0|kv|kv|0|" + kvm["kv"])
logId = STRING(kv["logid"])
sub = LOWERCASE(STRING(kv["subtype"]))

sensorType$ = "Fortinet FortiSandbox"
sensorEventId$ = IF logId != "" THEN logId ELSE NULL
hostId$ = STRING(kv["devid"])

# A syslog server gets date and time in UTC; tz and tzone name the FortiSandbox's own zone.
d = STRING(kv["date"])
t = STRING(kv["time"])
eventTimeUsec$ = IF d != "" AND t != "" THEN DATETIME(d + " " + t, "2006-01-02 15:04:05") ELSE eventTimeUsec$

vendor$ = "Fortinet"
provider$ = "FortiSandbox"
alertCategory$ = sub
level = LOWERCASE(STRING(kv["level"]))
risk = STRING(kv["risk"])
mname = STRING(kv["mname"])
fname = STRING(kv["fname"])
host = STRING(kv["host"])
attack = STRING(kv["attackname"])
botnet = STRING(kv["botnetname"])
url = STRING(kv["url"])
jobId = STRING(kv["jobid"])
alertId$ = IF jobId != "" THEN jobId ELSE NULL

# The title names the malware, the attack, the botnet or the URL's host.
what = ""
what = IF sub == "malware" THEN mname ELSE what
what = IF sub == "netattack" THEN attack ELSE what
what = IF sub == "netbotnet" AND botnet != "" THEN botnet ELSE what
what = IF sub == "netbotnet" AND botnet == "" THEN host ELSE what
what = IF sub == "neturl" THEN host ELSE what
kind = "FortiSandbox alert"
kind = IF sub == "malware" THEN "FortiSandbox malware" ELSE kind
kind = IF sub == "netattack" THEN "FortiSandbox network attack" ELSE kind
kind = IF sub == "netbotnet" THEN "FortiSandbox botnet connection" ELSE kind
kind = IF sub == "neturl" THEN "FortiSandbox malicious URL" ELSE kind
title$ = IF what != "" THEN kind + ": " + what ELSE kind
summary$ = IF sub == "malware" AND mname != "" AND fname != "" THEN mname + " in " + fname ELSE title$
summary$ = IF risk != "" THEN summary$ + " (" + risk + ")" ELSE summary$

# The rating (risk) sets the severity: Malicious, High Risk, Medium Risk, Low Risk or Clean.
# Without one, the log level does.
r = LOWERCASE(risk)
vendorSeverity$ = IF risk != "" THEN risk ELSE level
severity$ = "UNKNOWN_SEVERITY"
severity$ = IF level == "emergency" OR level == "alert" OR level == "critical" THEN "HIGH" ELSE severity$
severity$ = IF level == "error" OR level == "warning" THEN "MEDIUM" ELSE severity$
severity$ = IF level == "notice" THEN "LOW" ELSE severity$
severity$ = IF level == "information" OR level == "debug" THEN "INFO" ELSE severity$
severity$ = IF r == "malicious" THEN "CRITICAL" ELSE severity$
severity$ = IF r == "high risk" THEN "HIGH" ELSE severity$
severity$ = IF r == "medium risk" THEN "MEDIUM" ELSE severity$
severity$ = IF r == "low risk" THEN "LOW" ELSE severity$
severity$ = IF r == "clean" THEN "INFO" ELSE severity$

# Malware logs carry srcip, srcport, dstip and dstport; network alerts carry srcipport and
# dstipport as address:port.

dst = STRING(kv["dstip"])
sptText = STRING(kv["srcport"])
dptText = STRING(kv["dstport"])
sip = STRING(kv["srcipport"])
dip = STRING(kv["dstipport"])
sm = sip|^\[?(?P<ip>.+?)\]?:(?P<port>\d{1,5})$
dm = dip|^\[?(?P<ip>.+?)\]?:(?P<port>\d{1,5})$
src = IF src == "" THEN sm["ip"] ELSE src
dst = IF dst == "" THEN dm["ip"] ELSE dst
sptText = IF sptText == "" THEN sm["port"] ELSE sptText
dptText = IF dptText == "" THEN dm["port"] ELSE dptText
spt = sptText|^(?P<n>\d{1,5})$
dpt = dptText|^(?P<n>\d{1,5})$
sourceAddress$ = IF src != "" THEN src ELSE NULL
destinationAddress$ = IF dst != "" THEN dst ELSE NULL

destinationPort$ = IF dpt["n"] != "" THEN INT(dpt["n"], 10) ELSE NULL
destinationUrl$ = IF url != "" THEN url ELSE NULL
