# Source: Allied Telesis AlliedWare Plus firewall logs (facility local5), syslog to the data collector.
# Child of ALLIED_AWPLUS_SYSLOG_PARENT: firewall rule and filter lines, and connection logs, to Netflow.
#   Firewall rule 10: DENY IN=eth1 OUT=vlan1 [MAC=<dst>:<src>:<type>] SRC=... DST=... ... PROTO=TCP SPT=... DPT=...
#   END proto=TCP orig_src=... orig_dst=... orig_sport=... orig_dport=... orig_pkts=... orig_bytes=...
#     reply_src=... reply_dst=... reply_sport=... reply_dport=... reply_pkts=... reply_bytes=...
# Setting source_address and destination_address lets the IP Watchlist detector use these events.
# fw, conn, kv, kvText, eventType$, sensorType$, hostId$ and eventTimeUsec$ come from the parent.

isConn = eventType$ == "connection"

src = IF isConn THEN STRING(kv["orig_src"]) ELSE STRING(kv["SRC"])
dst = IF isConn THEN STRING(kv["orig_dst"]) ELSE STRING(kv["DST"])
sport = IF isConn THEN STRING(kv["orig_sport"]) ELSE STRING(kv["SPT"])
dport = IF isConn THEN STRING(kv["orig_dport"]) ELSE STRING(kv["DPT"])
sourceAddress$ = IF src != "" THEN src ELSE NULL
destinationAddress$ = IF dst != "" THEN dst ELSE NULL
sourcePort$ = IF sport != "" THEN INT(sport, 10) ELSE NULL
destinationPort$ = IF dport != "" THEN INT(dport, 10) ELSE NULL

# protocol takes the IANA number. The firewall logs a name; a connection log a name or a number.
proto = UPPERCASE(IF isConn THEN STRING(kv["proto"]) ELSE STRING(kv["PROTO"]))
pnum = proto|^(?P<n>\d+)$
protocol$ = IF pnum["n"] != "" THEN INT(pnum["n"], 10) ELSE NULL
protocol$ = IF proto == "TCP" THEN 6 ELSE protocol$
protocol$ = IF proto == "UDP" THEN 17 ELSE protocol$
protocol$ = IF proto == "ICMP" THEN 1 ELSE protocol$
protocol$ = IF proto == "ICMPV6" THEN 58 ELSE protocol$
protocol$ = IF proto == "GRE" THEN 47 ELSE protocol$
protocol$ = IF proto == "ESP" THEN 50 ELSE protocol$
protocol$ = IF proto == "AH" THEN 51 ELSE protocol$
protocol$ = IF proto == "SCTP" THEN 132 ELSE protocol$

# MAC= is the frame's destination MAC, source MAC and EtherType.
macText = STRING(kv["MAC"])
mac = macText|^(?P<d>(?:[0-9A-Fa-f]{2}:){5}[0-9A-Fa-f]{2}):(?P<s>(?:[0-9A-Fa-f]{2}:){5}[0-9A-Fa-f]{2})
sourceMac$ = IF mac["s"] != "" THEN mac["s"] ELSE NULL
destinationMac$ = IF mac["d"] != "" THEN mac["d"] ELSE NULL

# DENY and REJECT drop the packet (REJECT also answers the source), PERMIT passes it, and LOG only
# logs it, so its outcome is left unset. A tracked connection is one the firewall let through.
act = STRING(fw["act"])
flowAction$ = NULL
flowAction$ = IF act == "DENY" OR act == "REJECT" THEN "FW_BLOCKED" ELSE flowAction$
flowAction$ = IF act == "PERMIT" OR isConn THEN "FW_ALLOWED" ELSE flowAction$

# Connection logs: the originating side's packets and bytes went out, the reply side's came back.
# A reply address that differs from the original one shows NAT: the reply goes to the source NAT
# address, and comes from the real destination behind a port forward.
opk = STRING(kv["orig_pkts"])
oby = STRING(kv["orig_bytes"])
rpk = STRING(kv["reply_pkts"])
rby = STRING(kv["reply_bytes"])
txPacketCount$ = IF opk != "" THEN INT(opk, 10) ELSE NULL
txByteCount$ = IF oby != "" THEN INT(oby, 10) ELSE NULL
rxPacketCount$ = IF rpk != "" THEN INT(rpk, 10) ELSE NULL
rxByteCount$ = IF rby != "" THEN INT(rby, 10) ELSE NULL
rsrc = STRING(kv["reply_src"])
rdst = STRING(kv["reply_dst"])
rsport = STRING(kv["reply_sport"])
rdport = STRING(kv["reply_dport"])
snat = isConn AND rdst != "" AND rdst != src
dnat = isConn AND rsrc != "" AND rsrc != dst
sourceNatAddress$ = IF snat THEN rdst ELSE NULL
sourceNatPort$ = IF snat AND rdport != "" THEN INT(rdport, 10) ELSE NULL
destinationNatAddress$ = IF dnat THEN rsrc ELSE NULL
destinationNatPort$ = IF dnat AND rsport != "" THEN INT(rsport, 10) ELSE NULL

inIf = STRING(kv["IN"])
