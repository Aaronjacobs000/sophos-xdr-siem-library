addr = IF isSsh THEN STRING(w["addr"]) ELSE ""

targetUserName$ = IF user != "" THEN user ELSE NULL
sourceAddress$ = IF addr != "" AND IS_VALID_IP(addr) THEN addr ELSE NULL
targetHostName$ = IF host != "" THEN host ELSE NULL

atype = STRING(aaa["type"])
logonApplicationFamily$ = IF isSsh THEN "SSH" ELSE NULL

authSystem$ = IF isAaa AND svr != "" THEN svr ELSE "NEC UNIVERGE IX"
