# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: BeyondTrust Password Safe events forwarded by a BeyondInsight IBM QRadar connector in
# LEEF, over syslog to the data collector, pairs separated by tabs:
#   LEEF:1.0|BeyondTrust|BeyondInsight|<version>|<event ID>|cat=<category>	devTime=<time>
#   devTimeFormat=MMM dd yyyy HH:mm:ss	sev=<n>	src=<address>	usrName=<user>	...	AgentID=pbps
#   ...	UserName=<user>	RoleUsed=<role>	ObjectType=<type>	Operation=<operation>	Failed=<True|False>
#   Target=<what was acted on>	...
# BeyondTrust documents the LEEF layout with a Discovery Scanner event and lists the Password Safe
# keys in a table, but shows no Password Safe line, so the line layout is inferred.
# Password Safe audit records (category System: requests, approvals, password retrievals,
# sessions, account and system changes) and password change records (category Change) go to
# CloudAudit.

v = LEEF(originalData$)
cat = TRIM(STRING(v["Category"]))
cat = IF cat == "" THEN TRIM(STRING(v["cat"])) ELSE cat
isChange = LOWERCASE(cat) == "change"
objType = TRIM(STRING(v["ObjectType"]))
op = TRIM(STRING(v["Operation"]))

sensorType$ = "BeyondTrust Password Safe"
ver = TRIM(STRING(v["AgentVer"]))
sensorVersion$ = IF ver != "" THEN ver ELSE NULL
logId = TRIM(STRING(v["LogID"]))
logId = IF logId == "" THEN TRIM(STRING(v["LogSystemID"])) ELSE logId
sensorEventId$ = IF logId != "" THEN logId ELSE NULL

# devTime is UTC in the layout devTimeFormat names (MMM dd yyyy HH:mm:ss). Without it, LogTime,
# which BeyondTrust documents as MM/dd/yyyy HH:mm:ss, also read as UTC.
dev = TRIM(STRING(v["devTime"]))
devOk = dev|^[A-Z][a-z]{2} \d{2} \d{4} \d{2}:\d{2}:\d{2}$
lt = TRIM(STRING(v["LogTime"]))
ltOk = lt|^\d{2}/\d{2}/\d{4} \d{2}:\d{2}:\d{2}$
eventTimeUsec$ = IF ltOk[0] != "" THEN DATETIME(lt, "01/02/2006 15:04:05") ELSE eventTimeUsec$
eventTimeUsec$ = IF devOk[0] != "" THEN DATETIME(dev, "Jan 02 2006 15:04:05") ELSE eventTimeUsec$

eventSource$ = "BeyondTrust Password Safe"
eventType$ = IF cat != "" THEN cat ELSE NULL
# The event name is the object type and the operation, as BeyondTrust's Event Desc gives it
# (Functional Account Add); a password change record is a Password Change.
evName = objType + " " + op
evName = IF objType == "" OR op == "" THEN TRIM(STRING(v["EventDesc"])) ELSE evName
evName = IF isChange THEN "Password Change" ELSE evName
eventName$ = IF evName != "" THEN evName ELSE NULL

# LEEF doubles a backslash in a value, as in EXAMPLE\\jdoe.
user = TRIM(STRING(v["UserName"]))
user = IF user == "" THEN TRIM(STRING(v["usrName"])) ELSE user
user = REPLACE(user, `\\`, `\`)
userName$ = IF user != "" THEN user ELSE NULL
uid = TRIM(STRING(v["UserID"]))
userId$ = IF uid != "" THEN uid ELSE NULL
ip = TRIM(STRING(v["IPAddress"]))
sourceAddress$ = IF ip != "" THEN ip ELSE NULL

# Target names what was acted on, such as Asset:db01 MAccount:sa or FAccount=svc FDomain=example.
target = TRIM(STRING(v["Target"]))
asset = target|(?:^|\s)(?:Asset|System|MSystem):(?P<v>\S+)
acct = target|(?:^|\s)(?:MAccount|Account|FAccount)[:=](?P<v>\S+)
targetHostname$ = IF asset["v"] != "" THEN asset["v"] ELSE NULL
acctName = IF acct["v"] != "" THEN acct["v"] ELSE TRIM(STRING(v["AccountName"]))
targetUserName$ = IF acctName != "" THEN acctName ELSE NULL

# Failed is True or False on audit records; Result is S or F on password change records.
failed = LOWERCASE(TRIM(STRING(v["Failed"]))) == "true" OR TRIM(STRING(v["Result"])) == "F"
status$ = IF failed THEN "failed" ELSE "success"
comment = TRIM(STRING(v["Comment"]))
details = TRIM(STRING(v["Details"]))
errorMessage$ = IF failed AND comment != "" THEN comment ELSE NULL
errorMessage$ = IF failed AND comment == "" AND details != "" THEN details ELSE errorMessage$

lop = LOWERCASE(op)
readOnly$ = lop == "retrieve" OR lop == "read" OR lop == "view" OR lop == "download"
lot = LOWERCASE(objType)
workflow = lot == "password" OR lot == "release request" OR lot == "request" OR lot == "session"
managementEvent$ = isChange OR NOT (workflow OR readOnly$)

resources$ = []
obj.resourceId$ = TRIM(STRING(v["ObjectID"]))
obj.resourceType$ = objType
resources$ = IF obj.resourceId$ != "" THEN resources$ + [obj] ELSE resources$
ma.resourceId$ = TRIM(STRING(v["ManagedAccountID"]))
ma.resourceType$ = "Managed Account"
resources$ = IF ma.resourceId$ != "" THEN resources$ + [ma] ELSE resources$
ms.resourceId$ = TRIM(STRING(v["ManagedSystemID"]))
ms.resourceType$ = "Managed System"
resources$ = IF ms.resourceId$ != "" THEN resources$ + [ms] ELSE resources$

role = TRIM(STRING(v["RoleUsed"]))
d1.key$ = "role"

# Assign the list first: reading additionalEventData$.record$ before it is set fails the event.
additionalEventData$.record$ = []

d2.key$ = "object_type"

d4.key$ = "target"

d6.key$ = "request_id"

# ChangeReasonCd, with BeyondTrust's meaning for each letter.

reason = IF rc == "F" THEN "forced password reset" ELSE reason
reason = IF rc == "M" THEN "password reset on mismatch" ELSE reason
