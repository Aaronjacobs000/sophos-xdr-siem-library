# Standalone parser for the third-party alert schema: incident report and escalation events.
# Platform action and account notice events stay in the generic schema.
# Fields follow the Huntress Webhook Payload Reference (IncidentReportPayload, EscalationPayload).
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

sensorType$ = "Huntress"
vendor$ = "Huntress"
provider$ = "Huntress SOC"

eventType = json["$.event_type"]
logType$ = eventType
alertId$ = STRING(json["$.id"])
title$ = json["$.subject"]
status$ = json["$.status"]

report = CONTAINS(eventType, "incident_report.")
alertCategory$ = IF report THEN "Incident report" ELSE "Escalation"
escType = json["$.type"]
alertCategory$ = IF escType != "" AND escType != NULL THEN alertCategory$ + ": " + escType ELSE alertCategory$
sum = json["$.summary"]
summary$ = IF sum != "" AND sum != NULL THEN sum ELSE json["$.subject"]

# Huntress rates reports and escalations critical, high or low.
sev = LOWERCASE(json["$.severity"])
vendorSeverity$ = json["$.severity"]
severity$ = "UNKNOWN_SEVERITY"
severity$ = IF sev == "low" THEN "LOW" ELSE severity$
severity$ = IF sev == "medium" THEN "MEDIUM" ELSE severity$
severity$ = IF sev == "high" THEN "HIGH" ELSE severity$
severity$ = IF sev == "critical" THEN "CRITICAL" ELSE severity$

# Times are RFC 3339. The event time is when the report or escalation last changed.
layout = "2006-01-02T15:04:05.999999999Z07:00"
# Test the text, not the time: the engine can't compare a time value with NULL.
created = json["$.created_at"]
updated = json["$.updated_at"]
closed = json["$.closed_at"]
closed = IF closed == NULL OR closed == "" THEN json["$.resolved_at"] ELSE closed
createdTimeUsec$ = DATETIME(created, layout)
updatedTimeUsec$ = DATETIME(updated, layout)
closedTimeUsec$ = DATETIME(closed, layout)
eventTimeUsec$ = DATETIME(IF updated != NULL AND updated != "" THEN updated ELSE created, layout)
