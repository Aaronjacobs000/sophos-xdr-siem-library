# Source: Trend Micro Apex Central syslog forwarding, CEF, to the data collector:
#   CEF:0|Trend Micro|Apex Central|2019|<event id>|<name>|3|...
# Parent: recognizes Apex Central lines and hands them to its children by log type, from the
# event id prefix (AV:, PML:, BM:, CnC:, NCIE:, FH:) or the name. Logon, auditing, update and
# the other log types stay generic.

v = CEF(originalData$)
sig = STRING(v[4])
name = STRING(v[5])
prefix = sig|^(?P<p>AV|PML|BM|CnC|NCIE|FH|EAC|MS|WB):

# Variables and $ fields set here are visible to both children.
sensorType$ = "Trend Micro Apex Central"
eventType$ = "other"
eventType$ = IF prefix["p"] != "" THEN LOWERCASE(prefix["p"]) ELSE eventType$
eventType$ = IF sig == "Spyware Detected" THEN "spyware" ELSE eventType$
eventType$ = IF sig == "VAD" THEN "vad" ELSE eventType$
eventType$ = IF name == "Attack Discovery Detections" THEN "ade" ELSE eventType$

# rt is the event time in UTC: "Oct 13 2026 01:02:03 GMT+00:00" (some types write "GMT +00:00").
rt = REPLACE(STRING(v["rt"]), "GMT +", "GMT+")
eventTimeUsec$ = IF rt != "" THEN DATETIME(rt, "Jan _2 2006 15:04:05 GMT-07:00") ELSE eventTimeUsec$
host = STRING(v["TMCMLogDetectedHost"])
host = IF host == "" THEN STRING(v["dhost"]) ELSE host
host = IF host == "" THEN STRING(v["shost"]) ELSE host
hostId$ = IF host != "" THEN host ELSE NULL

# Critical threat type letters (reason), which can be a comma list.
reasons = JSON(`{"A":"Known Advanced Persistent Threat","B":"Social engineering attack","C":"Vulnerability attack","D":"Lateral movement","E":"Unknown threats","F":"C&C callback","G":"Ransomware"}`)
reason = STRING(v["reason"])
# The engine evaluates both sides of AND, so lookups sit behind a checked key in IF.
rkey = reason|^(?P<k>[A-G])$
reasonText = IF rkey["k"] != "" THEN STRING(reasons["$." + rkey["k"]]) ELSE reason
