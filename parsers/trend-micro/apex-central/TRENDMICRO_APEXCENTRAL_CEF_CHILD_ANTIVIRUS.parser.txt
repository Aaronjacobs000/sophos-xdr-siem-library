kind = eventType$
av = kind == "av"
pml = kind == "pml"
spy = kind == "spyware"

threatName$ = name
threatName$ = IF spy THEN v["cs1"] ELSE threatName$
threatName$ = IF pml THEN v["cs2"] ELSE threatName$
computerName$ = hostId$
userName$ = IF pml THEN v["suser"] ELSE v["duser"]
processName$ = IF pml THEN v["sproc"] ELSE NULL

# AV writes the folder in filePath and the file in fname; fname can also be a quarantine name
# (<SHA-1>;0x<CRC>), which isn't a path.
folder = STRING(v["filePath"])
file = STRING(v["fname"])
qname = file|^[0-9A-Fa-f]{40};0x[0-9A-Fa-f]+$
endsSlash = folder|\\$
filePath$ = folder
filePath$ = IF av AND endsSlash[0] != "" AND file != "" AND qname[0] == "" THEN folder + file ELSE filePath$
sha1 = STRING(v["fileHash"])
sha1 = IF sha1 == "" THEN STRING(v["filehash"]) ELSE sha1
sham = sha1|^(?P<h>[0-9A-Fa-f]{40})$
fileHash$.sha1$ = IF sham["h"] != "" THEN LOWERCASE(sham["h"]) ELSE NULL

# Predictive ML writes the action as a number from Trend's Action Mapping Table.
actions = JSON(`{"21":"File cleaned","22":"File deleted","23":"File quarantined","24":"File renamed","25":"File passed","26":"Unable to clean file. Passed","27":"Unable to clean file. # AV: the second action result says what finally happened when the first action failed.
second = STRING(v["cs6"])
result = act

result = IF spy THEN STRING(v["cs5"]) ELSE result
