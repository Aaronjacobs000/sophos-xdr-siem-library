#   CEF:0|TippingPoint|UnityOne|<version>|<filter number>|<filter name>|<severity>|app=... act=...
# Standalone parser for the NIDS schema: every event is an IPS or reputation filter hit.
# Setting source_address and destination_address lets the IP Watchlist detector use these events.
# The extension keys follow the SMS v4.2 column list: cs1 profile, cs2 policy UUID, cs3 signature
# UUID, cs4 zones, cs5 SMS name, cs6 filter message parameters, cn1 VLAN, cn2 taxonomy, cn3 packet
# trace, rt the event time in milliseconds. Keys with no value are left out.

v = CEF(originalData$)

sensorType$ = "Trend Micro TippingPoint"
device = STRING(v["dvchost"])

hostId$ = IF device != "" THEN device ELSE NULL

# rt is Unix time in milliseconds.
rtText = STRING(v["rt"])
rt = rtText|^(?P<ms>\d{10,13})$
eventTimeUsec$ = IF rt["ms"] != "" THEN INT(rt["ms"], 10) * 1000 ELSE eventTimeUsec$
