# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Nutanix Flow Network Security policy hit logs (Prism Central remote syslog, Security
# Policy Hit Logs module), sent by each AHV host:
#   INFO:<time>  [<policy UUID>] <policy name> [New|Update|Destroy] SRC= DST= PROTO= SPORT= DPORT=
#   ACTION= DIRECTION= ORIG: PKTS= BYTES= REPLY: PKTS= BYTES=
# Older releases leave out DIRECTION and tag the lines flow-hitCountN instead of Genesis_N.
# Standalone parser for the Netflow schema. Setting source_address and destination_address lets
# the IP Watchlist detector use these events.

hdr = originalData$|^(?:<\d{1,3}>)?(?:\d{4}-\d{2}-\d{2}T\S+|[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?P<host>\S+)\s+(?P<tag>[^\s:\[]+)
m = originalData$|INFO:(?P<time>\d{4}/\d{2}/\d{2}\s+\d{2}:\d{2}:\d{2})\s+\[(?P<puuid>[0-9A-Fa-f-]{36})\]\s+(?P<policy>.*?)\s+\[(?P<state>[A-Za-z]+)\]\s+SRC=(?P<src>\S+)\s+DST=(?P<dst>\S+)\s+PROTO=(?P<proto>\S+)(?:\s+SPORT=(?P<sport>\d+))?(?:\s+DPORT=(?P<dport>\d+))?\s+ACTION=(?P<action>[A-Za-z]+)(?:\s+DIRECTION=(?P<dir>[A-Za-z]+))?(?:.*?ORIG:\s+PKTS=(?P<opk>\d+)\s+BYTES=(?P<oby>\d+)\s+REPLY:\s+PKTS=(?P<rpk>\d+)\s+BYTES=(?P<rby>\d+))?

sensorType$ = "Nutanix Flow Network Security"
hostId$ = hdr["host"]
# The hit time has no zone. AHV hosts keep UTC, so it is read as UTC.
eventTimeUsec$ = DATETIME(m["time"], "2006/01/02 15:04:05")

sourceAddress$ = m["src"]
destinationAddress$ = m["dst"]
sourcePort$ = m["sport"]
destinationPort$ = m["dport"]

# protocol takes the IANA number; the log has a name.
proto = UPPERCASE(m["proto"])
protocol$ = NULL
protocol$ = IF proto == "TCP" THEN 6 ELSE protocol$
protocol$ = IF proto == "UDP" THEN 17 ELSE protocol$
protocol$ = IF proto == "ICMP" THEN 1 ELSE protocol$
protocol$ = IF proto == "ICMPV6" THEN 58 ELSE protocol$

# ALLOW and DROP are enforced. MONITOR is a policy in monitor mode, which lets the traffic through.
action = UPPERCASE(m["action"])
flowAction$ = NULL
flowAction$ = IF action == "ALLOW" OR action == "MONITOR" THEN "FW_ALLOWED" ELSE flowAction$
flowAction$ = IF action == "DROP" THEN "FW_BLOCKED" ELSE flowAction$

dir = UPPERCASE(m["dir"])
direction$ = NULL
direction$ = IF dir == "INBOUND" THEN "INBOUND" ELSE direction$
direction$ = IF dir == "OUTBOUND" THEN "OUTBOUND" ELSE direction$

# ORIG counts the source's packets and bytes, REPLY the destination's, for the session so far.
txPacketCount$ = m["opk"]
txByteCount$ = m["oby"]
rxPacketCount$ = m["rpk"]
rxByteCount$ = m["rby"]

r1.key$ = "policy_name"
r1.value$ = m["policy"]
r2.key$ = "policy_uuid"
r2.value$ = m["puuid"]
# New is the first packet of a session, Update repeats about every 30 seconds while it is open,
# and Destroy is its end, with the final counts.
r3.key$ = "session_state"
r3.value$ = m["state"]
r4.key$ = "policy_action"
r4.value$ = m["action"]
r5.key$ = "syslog_tag"
r5.value$ = hdr["tag"]
eventMetadata$.record$ = [r1, r2, r3, r4, r5]
