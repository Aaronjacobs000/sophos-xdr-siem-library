#     LogoutReason=...,] UserName=..., UserAddress=..., LocalAddress=..., VPNInstanceName=...)
#   SSH/4/SSH_FAIL (V200): (IP=..., VpnInstanceName=..., UserName=..., Times=..., FailedReason=...)
#   AAA/6/LOCALACCOUNT_LOCK and UNLOCK: Local account <account> has been locked|unlocked.(UserIp=..., OperatorName=...)
# Huawei writes ** when it can't get the user or address, which is left unset.
# mod, brief, desc, kv, host, eventType$, sensorType$, sensorEventId$, hostId$ and eventTimeUsec$ come from the parent.

acct = desc|^Local account (?P<a>.+?) has been (?P<state>locked|unlocked)
isAaa = mod == "AAA"
user = STRING(kv["UserName"])
user = IF isAaa THEN STRING(acct["a"]) ELSE user
targetUserName$ = IF user != "" AND user != "**" THEN user ELSE NULL
operator = STRING(kv["OperatorName"])
sourceUserName$ = IF isAaa AND operator != "" THEN operator ELSE NULL
ip = STRING(kv["Ip"])
ip = IF ip == "" THEN STRING(kv["RemoteIp"]) ELSE ip
ip = IF ip == "" THEN STRING(kv["UserAddress"]) ELSE ip
ip = IF ip == "" THEN STRING(kv["IP"]) ELSE ip
ip = IF ip == "" THEN STRING(kv["UserIp"]) ELSE ip
sourceAddress$ = IF ip != "" AND ip != "**" THEN ip ELSE NULL
local = STRING(kv["LocalIp"])
local = IF local == "" THEN STRING(kv["LocalAddress"]) ELSE local
targetAddress$ = IF local != "" AND local != "**" THEN local ELSE NULL
targetHostName$ = IF host != "" THEN host ELSE NULL

# The access channel: SSH for the SSH module, else the user or access type the device logged.
kind = STRING(kv["UserType"])
kind = IF kind == "" THEN STRING(kv["AccessType"]) ELSE kind
kind = IF mod == "SSH" THEN "SSH" ELSE kind
kind = IF isAaa THEN "local account" ELSE kind
logonApplicationFamily$ = IF kind != "" THEN kind ELSE NULL
authSystem$ = IF kind != "" THEN "Huawei VRP " + kind ELSE "Huawei VRP"

logout = brief == "LOGOUT" OR brief == "SSH_USER_LOGOUT"
locked = brief == "LOCALACCOUNT_LOCK"
unlocked = brief == "LOCALACCOUNT_UNLOCK"
success = brief == "LOGIN" OR brief == "SSH_USER_LOGIN" OR logout OR unlocked
authCategory$ = "ACCOUNT_LOGON"
authCategory$ = IF logout THEN "ACCOUNT_LOGOFF" ELSE authCategory$
authCategory$ = IF locked THEN "ACCOUNT_LOCK" ELSE authCategory$
authCategory$ = IF unlocked THEN "ACCOUNT_UNLOCK" ELSE authCategory$
action$ = IF success THEN "LOGON" ELSE "FAILURE"
action$ = IF logout THEN "LOGOFF" ELSE action$
action$ = IF locked OR unlocked THEN "ACCOUNT" ELSE action$
actionResult$ = IF success THEN "SUCCESS" ELSE "FAILED"
why = STRING(kv["Reason"])
why = IF why == "" THEN STRING(kv["FailedReason"]) ELSE why
why = IF why == "" AND brief == "LOGIN_FAIL_FOR_INPUT_TIMEOUT" THEN "Input timed out" ELSE why
why = IF locked THEN "Local account locked" ELSE why
extraFailurereason$ = IF NOT success AND why != "" THEN why ELSE NULL
lwhy = LOWERCASE(why)
failureCategory$ = IF success THEN NULL ELSE "OTHER_ERROR"
failureCategory$ = IF NOT success AND (CONTAINS(lwhy, "password") OR CONTAINS(lwhy, "authentication")) THEN "INCORRECT_USER_OR_PASSWORD" ELSE failureCategory$
# Huawei gives a wrong user name or password as the cause of SHELL/4/LOGINFAILED, which has no reason field.
failureCategory$ = IF mod == "SHELL" AND brief == "LOGINFAILED" THEN "INCORRECT_USER_OR_PASSWORD" ELSE failureCategory$
failureCategory$ = IF locked THEN "ACCOUNT_LOCKED_OUT" ELSE failureCategory$

# The line, VPN, failure count, sign-in method and SSH service go to event_metadata.
lineM = desc|(?:logging in to|logging out of)\s+(?P<l>\S+?)\.?$
