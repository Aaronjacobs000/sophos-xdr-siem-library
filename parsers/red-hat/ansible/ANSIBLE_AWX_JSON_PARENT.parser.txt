#    "cluster_host_id": "<node>", "tower_uuid": "<install>"}
# Parent: claims every record from an awx logger and hands them on:
#   awx.api.generics sign-ins, failed sign-ins and sign-outs        to the Auth child
#   awx.analytics.activity_stream (the audit trail)                 to the Management Event child
#   awx.analytics.job_events host results (runner_on_ok, _failed,

jm = originalData$|(?P<j>\{"@timestamp":.*\})\s*$
j = JSON(jm["j"])
logger = STRING(j["$.logger_name"])
msg = STRING(j["$.message"])
node = STRING(j["$.cluster_host_id"])

# Variables and $ fields set here are visible to every child.
sensorType$ = "Ansible AWX"
hostId$ = IF node != "" THEN node ELSE NULL
# The key starts with @, which Sophos NG-SIEM reads only in brackets.
eventTimeUsec$ = DATETIME(j["$['@timestamp']"], "2006-01-02T15:04:05.999999999Z07:00")

event = STRING(j["$.event"])
hostResult = event == "runner_on_ok" OR event == "runner_on_failed" OR event == "runner_on_unreachable" OR event == "runner_on_async_failed" OR event == "runner_item_on_ok" OR event == "runner_item_on_failed"
lf = msg|^(?P<f>Login failed for user )

eventType$ = "other"
eventType$ = IF signIn THEN "auth" ELSE eventType$
eventType$ = IF logger == "awx.analytics.activity_stream" THEN "activity" ELSE eventType$
eventType$ = IF logger == "awx.analytics.job_events" AND hostResult THEN "job" ELSE eventType$
