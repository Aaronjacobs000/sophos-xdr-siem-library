# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-ngsiem-parser-studio
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: N-able N-central Audit Export over syslog.
# Child of NCENTRAL_AUDIT_SYSLOG_PARENT: console logins (local and IdP), logouts and failed
# logins to Auth. A record is a failure when its action text says "fail".
# h, msg, msgid, user, srcIp, action, lowAction, eventType$, sensorType$ and eventTimeUsec$ come from the parent.

targetUserName$ = user
sourceAddress$ = srcIp
authSystem$ = "N-central " + msgid
idp = msg|using (?P<idp>.+?)(?:,|$)
authSystem$ = IF idp["idp"] != "" THEN "N-central " + idp["idp"] ELSE authSystem$

failed = CONTAINS(lowAction, "fail")
logout = CONTAINS(lowAction, "logged out") OR CONTAINS(LOWERCASE(msgid), "logout")
authCategory$ = IF logout THEN "ACCOUNT_LOGOFF" ELSE "ACCOUNT_LOGON"
action$ = "LOGON"
action$ = IF logout THEN "LOGOFF" ELSE action$
action$ = IF failed THEN "FAILURE" ELSE action$
actionResult$ = IF failed THEN "FAILED" ELSE "SUCCESS"
# The audit record doesn't say why a login failed.
failureCategory$ = IF failed THEN "OTHER_ERROR" ELSE NULL

m1.key$ = "audit_event"
m1.value$ = msgid
m2.key$ = "action"
m2.value$ = action
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [m1, m2]
cp = msg|customer path is \[(?P<path>[^\]]*)\]
m3.key$ = "customer_path"
m3.value$ = cp["path"]
eventMetadata$.record$ = IF cp["path"] != "" THEN eventMetadata$.record$ + [m3] ELSE eventMetadata$.record$
