# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-ngsiem-parser-studio
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: HPE Aruba Networking EdgeConnect SD-WAN appliance logs (ECOS), sent by the Orchestrator
# Logging template's remote log receivers to the data collector.
# Parent: recognizes firewall policy logs (netflowd), stateful WAN drop logs (tunneld) and the
# appliance audit log (mgmtd), in text form and, for the firewall, in the Jsonify JSON form.
#   Firewall policy and WAN drop logs   to the Netflow child
#   Audit log                           to the Management Event child
# System, alarm, DDoS threshold and IDS/IPS lines aren't claimed.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from the JSON form.

# ECOS writes the year after the time: Oct  4 05:17:28 2021 <appliance> netflowd[10239]: ...
# The JSON form has no program name: <time> <host> {"logType":"Zone Based Firewall",...}
hdr = originalData$|^(?:<\d{1,3}>)?(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2}(?:\s+\d{4})?)\s+(?P<host>\S+)\s+(?P<rest>.*)$
rest = hdr["rest"]
prog = rest|^(?P<name>[\w.-]+)\[\d+\]:\s*(?P<msg>.*)$
msg = prog["msg"]

jsonMatch = originalData$|(\{".*})\s*$
isJson = jsonMatch[1] != ""
json = JSON(IF isJson THEN jsonMatch[1] ELSE "{}")
logType = STRING(json["$.logType"])

# Text firewall lines are comma-separated Key:Value pairs after the level, for example
# [NOTICE] Action:allow, Reason:flow create, ... WAN drop lines start at LogType:.
fwText = msg|^\[(?P<level>[A-Z]+)\]\s+(?P<kv>Action:.*)$
wanText = rest|\[(?P<level>[A-Z]+)\]:?\s*LogType:Stateful WAN Drops,(?P<kv>.*)$
kvText = IF fwText["kv"] != "" THEN fwText["kv"] ELSE wanText["kv"]
kv = SPLIT_NAME_VALUES(STRING(kvText), ",", ":", `"`)
wanDrop = wanText["kv"] != "" OR logType == "Stateful WAN Drops"
level = IF isJson THEN STRING(json["$.level"]) ELSE STRING(fwText["level"])
level = IF level != "" THEN level ELSE STRING(wanText["level"])

# Audit lines: <user>@<address or -> action <category> <operation> <status> params:<name=value ,...>
audit = msg|^(?P<user>[^\s@]+)@(?P<from>\S+)\s+action\s+(?P<text>\S+\s+(?P<op>\S+)\s+(?P<status>\S+)(?:\s+params:.*)?)$

# Variables and $ fields set here are visible to both children.
sensorType$ = "HPE Aruba EdgeConnect"
eventType$ = "other"
eventType$ = IF fwText["kv"] != "" OR wanText["kv"] != "" OR logType == "Zone Based Firewall" OR logType == "Stateful WAN Drops" THEN "firewall" ELSE eventType$
eventType$ = IF prog["name"] == "mgmtd" AND audit["op"] != "" THEN "audit" ELSE eventType$

# The appliance name: hostname in the JSON form, the header host otherwise.
jsonHost = STRING(json["$.hostname"])
appliance = IF jsonHost != "" THEN jsonHost ELSE hdr["host"]
hostId$ = appliance
sensorId$ = appliance

# The JSON form carries its own timestamp with the year. ECOS doesn't document the time zone, so
# the time is read as UTC; set the appliances to UTC.
jsonTime = STRING(json["$.timestamp"])
time = IF jsonTime != "" THEN jsonTime ELSE hdr["time"]
year = time|\s\d{4}$
eventTimeUsec$ = IF year[0] != "" THEN DATETIME(time, "Jan _2 15:04:05 2006") ELSE DATETIME(time, "Jan _2 15:04:05", TRUE)
