# Parent: hands policy events (Configuration, SCADA, Network Threats and Network events) to the
# Thirdparty child and system log events (Tenable.ot, class 9999) to the CloudAudit child. Keep
# alive messages (class 8888) stay generic.

v = CEF(originalData$)
classId = STRING(v[4])
name = STRING(v[5])

# Variables and $ fields set here are visible to both children.
sensorType$ = "Tenable OT Security"
eventType$ = "alert"

eventType$ = IF classId == "8888" THEN "keepalive" ELSE eventType$

# rt is when the appliance registered the event, as MMM D YYYY HH:mm:ss with no zone, read as UTC.
# System events have no rt, so their time is the syslog header's, which has no year.
