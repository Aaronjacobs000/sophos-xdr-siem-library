# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Omnissa Unified Access Gateway (UAG) syslog, from Advanced Settings > System Configuration
# > Syslog in the Admin UI, in the formats of Omnissa's "Syslog Formats and Events" page:
#   audit.log, admin.log:  <time> <host> <app> [<thread>]<level> <class>[<function>: <line>] - <message>
#   esmanager.log:         ... [<function>: <line>][<client IP>][<user>][<session type>][<session id>] - <message>
# The app is uag-admin_uag-audit (admin audit events), uag-admin_ (admin system events) or
# uag-esmanager_ (edge services, such as Horizon); UAG 3.x wrote UAG-ESMANAGER and only the session
# ID bracket.
# Parent: recognizes every UAG line in these formats and hands on:
#   LOGIN_SUCCESS, LOGIN_FAILED, LOGOUT_SUCCESS (Admin UI)                               to Auth
#   Authentication successful for user, AUTH FAILED for user, Authentication attempt -
#   FAILED, HORIZON_SESSION:AUTHENTICATED and HORIZON_SESSION:TERMINATED (edge services)  to Auth
#   CONFIG_CHANGE (audit) and <area>:CONFIG_CHANGED (admin)                              to ManagementEvent
# Other edge service and monitoring lines stay in generic. Root console and SSH logins arrive as
# login and sshd lines, which built-in parsers read.

hdr = originalData$|^(?:<\d{1,3}>)?(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?P<host>\S+)\s+(?P<app>[Uu][Aa][Gg]-[A-Za-z_-]+):\s
m = originalData$|\s[Uu][Aa][Gg]-[A-Za-z_-]+:\s\[(?P<thread>[^\]]*)\](?P<level>TRACE|DEBUG|INFO|WARN|ERROR|FATAL)\s+(?P<class>\S+)\[(?P<func>[^\]:]+): (?P<lineno>\d+)\](?P<brackets>(?:\[[^\]]*\])?(?:\[[^\]]*\])?(?:\[[^\]]*\])?(?:\[[^\]]*\])?(?:\[[^\]]*\])?) - (?P<msg>.*?)\s*$
# Edge service lines: four brackets (client IP, user, session type, session ID), or the session ID alone in UAG 3.x.
es = m["brackets"]|^(?:\[(?P<cip>[^\]]*)\]\[(?P<user>[^\]]*)\]\[(?P<stype>[^\]]*)\])?\[(?P<sid>[^\]]*)\]$
msg = m["msg"]
# Audit events: <EVENT>: SOURCE_IP_ADDR=<address> USERNAME=<user>[: REASON=<text> | : CHANGE=<changes>]
audit = msg|^(?P<event>[A-Z_]+): SOURCE_IP_ADDR=(?P<ip>[^\s:]+):? USERNAME=(?P<user>.+?)(?:: (?:REASON=(?P<reason>.*)|CHANGE=(?P<change>.*)|INFO=(?P<info>.*)))?$
# Admin system events: <AREA>:CONFIG_CHANGED:<changes>
settings = msg|^(?P<area>[A-Z][A-Z_]*):CONFIG_CHANGED:(?P<change>.*)$

# Variables and $ fields set here are visible to both children.
sensorType$ = "Omnissa Unified Access Gateway"
uag = hdr["host"]
hostId$ = uag
# A BSD syslog time has no year and no zone: DATETIME picks the year (TRUE) and reads UTC.
eventTimeUsec$ = DATETIME(hdr["time"], "Jan _2 15:04:05", TRUE)

event = audit["event"]
eventType$ = "other"
eventType$ = IF event == "LOGIN_SUCCESS" OR event == "LOGIN_FAILED" OR event == "LOGOUT_SUCCESS" THEN "auth" ELSE eventType$
eventType$ = IF event == "CONFIG_CHANGE" OR settings["area"] != "" THEN "mgmt" ELSE eventType$
edge = msg|^(?P<kind>Authentication successful for user |AUTH FAILED for user |Authentication attempt - FAILED$|HORIZON_SESSION:AUTHENTICATED|HORIZON_SESSION:TERMINATED)
eventType$ = IF edge["kind"] != "" THEN "auth" ELSE eventType$
