!CONFIRMSTRING=%UCSM-[0-7]-AUDIT:\s*\[session\]\[[^\]]*\]\[creation\].*\]\s*(?:Web|Fabric)\s+[AB]:\s+remote user\s+\S+\s+logged in from\s

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Cisco UCS Manager syslog from the fabric interconnects, with audits turned on as a source
# (enable syslog source audits):
#   <PRI>: <year> <Mon> <day> <hh:mm:ss> <zone>: %UCSM-<severity>-<MNEMONIC>: <body>
#   AUDIT bodies: [trigger][user][change][] <description>, or on newer releases
#   [trigger][user][change][session][record][affected DN][change set][] <description>
# Child of the global parser Cisco_IOS_Master, which claims %UCSM lines and leaves them without a
# schema. In Sophos Fusion, turn on Set Global Parent in Parser Script. The parser reads the whole
# line itself, not the parent's variables.
# Administrator sign-ins to UCS Manager to Auth: AUDIT [session] creation records,
#   Web A|B: remote user <user> logged in from <address>      (the UCS Manager web interface)
#   Fabric A|B: remote user <user> logged in from <address>   (SSH to a fabric interconnect)
# Failed sign-ins are %AUTHPRIV-3-SYSTEM_MSG pam_aaa lines, which the built-in Cisco IOS parsers
# already read to Auth.

# The time carries the zone the fabric interconnects are set to, as a name. The common names below
# are read with their offset (CST as US Central); any other name is read as UTC, so keep the
# fabric interconnects on UTC or one of these zones.
hm = originalData$|^(?:<\d{1,3}>)?:?\s*(?:(?P<host>[^\s:]+)\s*:\s*)?(?P<t>\d{4}\s+[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})(?:\.\d+)?\s*(?P<tz>[A-Za-z]{1,6})?\s*:\s*%UCSM-(?P<sev>[0-7])-(?P<mn>[A-Z0-9_]+):\s*(?P<body>.*?)\s*$
body = STRING(hm["body"])
host = STRING(hm["host"])
tzn = UPPERCASE(STRING(hm["tz"]))
off = "+0000"
off = IF tzn == "AEST" THEN "+1000" ELSE off
off = IF tzn == "AEDT" THEN "+1100" ELSE off
off = IF tzn == "ACST" THEN "+0930" ELSE off
off = IF tzn == "ACDT" THEN "+1030" ELSE off
off = IF tzn == "AWST" THEN "+0800" ELSE off
off = IF tzn == "NZST" THEN "+1200" ELSE off
off = IF tzn == "NZDT" THEN "+1300" ELSE off
off = IF tzn == "JST" THEN "+0900" ELSE off
off = IF tzn == "KST" THEN "+0900" ELSE off
off = IF tzn == "SGT" THEN "+0800" ELSE off
off = IF tzn == "HKT" THEN "+0800" ELSE off
off = IF tzn == "CET" THEN "+0100" ELSE off
off = IF tzn == "CEST" THEN "+0200" ELSE off
off = IF tzn == "EET" THEN "+0200" ELSE off
off = IF tzn == "EEST" THEN "+0300" ELSE off
off = IF tzn == "WEST" THEN "+0100" ELSE off
off = IF tzn == "BST" THEN "+0100" ELSE off
off = IF tzn == "EST" THEN "-0500" ELSE off
off = IF tzn == "EDT" THEN "-0400" ELSE off
off = IF tzn == "CST" THEN "-0600" ELSE off
off = IF tzn == "CDT" THEN "-0500" ELSE off
off = IF tzn == "MST" THEN "-0700" ELSE off
off = IF tzn == "MDT" THEN "-0600" ELSE off
off = IF tzn == "PST" THEN "-0800" ELSE off
off = IF tzn == "PDT" THEN "-0700" ELSE off
eventTimeUsec$ = IF STRING(hm["t"]) != "" THEN DATETIME(hm["t"] + " " + off, "2006 Jan _2 15:04:05 -0700") ELSE eventTimeUsec$

sensorType$ = "Cisco UCS Manager"
sensorEventId$ = "UCSM-" + hm["sev"] + "-" + hm["mn"]
hostId$ = IF host != "" THEN host ELSE NULL

a = body|^\[(?P<trig>[^\]]*)\]\[(?P<user>[^\]]*)\]\[(?P<ind>[^\]]*)\](?:\[(?P<sess>[^\]]*)\]\[(?P<tx>[^\]]*)\]\[(?P<dn>[^\]]*)\]\[(?P<cs>[^\]]*)\])?\[\]\s*(?P<desc>.*)$
sess = STRING(a["sess"])
tx = STRING(a["tx"])
dn = STRING(a["dn"])
desc = STRING(a["desc"])

li = desc|^(?P<via>Web|Fabric)\s+(?P<fi>[AB]):\s+remote user\s+(?P<u>\S+)\s+logged in from\s+(?P<ip>\S+)
targetUserName$ = li["u"]
ip = STRING(li["ip"])
sourceAddress$ = IF IS_VALID_IP(ip) THEN ip ELSE NULL
targetHostName$ = IF host != "" THEN host ELSE NULL
authSystem$ = "Cisco UCS Manager"
logonApplicationFamily$ = IF li["via"] == "Web" THEN "Web" ELSE "SSH"
# The session is id: in the change set on newer releases (id:web_40207_B, name:<user>, ...).
cs = STRING(a["cs"])
sid = cs|(?:^|[\s,])id:(?P<id>[^,\s]+)
session = STRING(sid["id"])
sessionId$ = IF session != "" THEN session ELSE NULL
authCategory$ = "ACCOUNT_LOGON"
action$ = "LOGON"
actionResult$ = "SUCCESS"

# The fabric interconnect, the audit record and the session object go to event_metadata.
m1.key$ = "fabric_interconnect"
m1.value$ = li["fi"]
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [m1]
m2.key$ = "audit_record"
m2.value$ = tx
eventMetadata$.record$ = IF tx != "" THEN eventMetadata$.record$ + [m2] ELSE eventMetadata$.record$
m3.key$ = "affected_dn"
m3.value$ = dn
eventMetadata$.record$ = IF dn != "" THEN eventMetadata$.record$ + [m3] ELSE eventMetadata$.record$
