!SAMPLE=<37>Oct 20 22:24:05 c220-node01 AUDIT[2257]: CIMC Locator LED is modified to "ON" by User:admin from Interface:redfish Remote IP:192.0.2.25
!CONFIRMWITH=PATTERN
!CONFIRMSTRING=\sAUDIT\[\d+\]:\s.*\sby User:\S*\s+from Interface:\S*(?:\s+Remote IP:\S*)?\s*$
!SCHEMA=scwx.managementevent

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Cisco Integrated Management Controller (CIMC, the standalone or Intersight managed server
# controller) audit records, sent by remote syslog at facility auth:
#   <PRI><Mon> <day> <hh:mm:ss> <host> AUDIT[<pid>]: <change> by User:<user> from Interface:<interface> [Remote IP:<address>]
# Standalone parser for Management Event: the change as the command, the user (none for (null),
# which Intersight's own changes show), the interface (such as redfish) and the client address.

c = originalData$|^(?:<\d{1,3}>)?(?P<t>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?:(?P<host>\S+)\s+)?AUDIT\[(?P<pid>\d+)\]:\s*(?P<desc>.*?)\s+by User:(?P<user>\S*)\s+from Interface:(?P<iface>\S*)(?:\s+Remote IP:(?P<ip>\S*))?\s*$
sensorType$ = "Cisco IMC"
sensorEventId$ = "AUDIT"
host = STRING(c["host"])
hostId$ = IF host != "" THEN host ELSE NULL
targetHostname$ = IF host != "" THEN host ELSE NULL
# The time has no year or zone; it is read as UTC in the current year. The controller's time zone is
# UTC unless it was changed.
eventTimeUsec$ = DATETIME(c["t"], "Jan _2 15:04:05", TRUE)

user = STRING(c["user"])
username$ = IF user != "" AND user != "(null)" THEN user ELSE NULL
iface = STRING(c["iface"])
channel$ = IF iface != "" THEN iface ELSE NULL
ip = STRING(c["ip"])
sourceAddress$ = IF IS_VALID_IP(ip) THEN ip ELSE NULL
command$ = c["desc"]
result$ = "success"
