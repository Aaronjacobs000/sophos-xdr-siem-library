# Child of CISCO_SDWAN_SYSLOG_PARENT: each one to Antivirus, with the malware name, the file's SHA-256
# and name, and Drop as blocked. The disposition, file type, addresses, ports, System_IP and policy go
# to event_metadata. Filename can be empty.

threatType$ = "FILE"
disp = amp["disp"]

act = utd["act"]

policyName$ = IF pol["v"] != "" THEN pol["v"] ELSE NULL
