# Standalone parser for the Email schema. Text mail logs (mail_logs) are not covered.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from the Python-style values ESA writes.
# The admin picks which fields the subscription writes and their order, so every key is optional.

v = CEF(originalData$)
sensorType$ = "Cisco Secure Email Gateway"
sensorVersion$ = v[3]
# The appliance serial number identifies the gateway.
hostId$ = v["deviceExternalId"]

# Event time: the delivery (DCID) time, else the connection (ICID) time, else the syslog header.
# AsyncOS 14 and earlier write endTime and startTime. ESA writes the appliance's local time
# with no zone, so it is read as UTC.
t = STRING(v["end"])
t = IF t == "" THEN STRING(v["endTime"]) ELSE t
t = IF t == "" THEN STRING(v["EndTime"]) ELSE t
t = IF t == "" THEN STRING(v["start"]) ELSE t
t = IF t == "" THEN STRING(v["startTime"]) ELSE t
hdr = originalData$|^(?:<\d{1,3}>)?(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s
eventTimeUsec$ = IF t != "" THEN DATETIME(t, "Mon Jan _2 15:04:05 2006") ELSE eventTimeUsec$
eventTimeUsec$ = IF t == "" AND hdr["time"] != "" THEN DATETIME(hdr["time"], "Jan _2 15:04:05", TRUE) ELSE eventTimeUsec$

# suser is the envelope sender (Mail From); ESAFriendlyFrom, the header From, goes to event_metadata.
fromEmailAddress$ = v["suser"]
rcpt = STRING(v["duser"])
toEmailAddress$ = IF rcpt != "" THEN SPLIT(rcpt, ";") ELSE NULL

# ESA quotes the subject and Message-ID Python style: '...', or "..." when the value has a '.
subj = STRING(v["msg"])
sq = subj|^'(?P<s>.*)'$
dq = subj|^"(?P<s>.*)"$
subject$ = v["msg"]
subject$ = IF dq["s"] != "" THEN dq["s"] ELSE subject$
subject$ = IF sq["s"] != "" THEN sq["s"] ELSE subject$
mid = STRING(v["ExternalMsgID"])
midm = mid|^['"]?<?(?P<id>[^'"<>]+)>?['"]?$
messageId$ = midm["id"]

size = STRING(v["ESAMsgSize"])
messageSize$ = IF size != "" THEN INT(size, 10) ELSE NULL
senderIp$ = v["src"]
senderIp$ = IF STRING(v["src"]) == "" THEN v["sourceAddress"] ELSE senderIp$

dir = STRING(v["deviceDirection"])
direction$ = NULL
direction$ = IF dir == "0" THEN "INBOUND" ELSE direction$
direction$ = IF dir == "1" THEN "OUTBOUND" ELSE direction$

# act is the final action: DELIVERED, QUARANTINED, DQ (delayed quarantine), DROPPED or BOUNCED.
act = UPPERCASE(STRING(v["act"]))
details = STRING(v["ESAFinalActionDetails"])
status$ = NULL
status$ = IF act == "DELIVERED" THEN "DELIVERED" ELSE status$
status$ = IF act == "QUARANTINED" OR act == "DQ" THEN "QUARANTINED" ELSE status$
status$ = IF act == "DROPPED" OR act == "BOUNCED" THEN "BLOCKED" ELSE status$
quarantineReason$ = IF status$ == "QUARANTINED" AND details != "" THEN details ELSE NULL

# The first attachment's name and SHA-256 come from ESAAttachmentDetails, a Python-style dict:
#   {'<file name>': {'AMP': {'Verdict': '...', 'fileHash': '<sha256>'}, 'BodyScanner': {'fsize': <bytes>}}}
attText = STRING(v["ESAAttachmentDetails"])
attName = attText|^\{'(?P<name>[^']+)':

a1.fileSize$ = IF attSize["size"] != "" THEN INT(attSize["size"], 10) ELSE NULL
# A separate record with the hash, so a file with no hash has no empty file_hash.
a2.fileName$ = attName["name"]
a2.fileSize$ = a1.fileSize$
a2.fileHash$.sha256$ = attSha["sha"]
attList = []

dlp = UPPERCASE(STRING(v["ESADLPVerdict"]))

t_dlp.name$ = "DLP"
