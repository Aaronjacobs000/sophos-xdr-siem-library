!PARENT=Cisco_IOS_Master
!CONFIRMWITH=PATTERN
!CONFIRMSTRING=%UC_AUDITLOG-\d-(?:Administrative|Security|Critical)Event\s?:\s*%\[.*\[\s*EventType\s*[=]\s*(?:[^U\]\s][^\]]*|U(?:[^s\]\s][^\]]*)?|Us(?:[^e\]\s][^\]]*)?|Use(?:[^r\]\s][^\]]*)?|User(?:[^L\]\s][^\]]*)?|UserL(?:[^o\]\s][^\]]*)?|UserLo(?:[^g\]\s][^\]]*)?|UserLog(?:[^g\]\s][^\]]*)?|UserLogg(?:[^i\]\s][^\]]*)?|UserLoggi(?:[^n\]\s][^\]]*)?|UserLoggin(?:[^g\]\s][^\]]*)?|UserLogging[^\]\s][^\]]*)\]|%?CCM_TOMCAT-GENERIC-\d-AuditEventGenerated\s?:\s*Audit Event Generated\s
!SCHEMA=scwx.managementevent

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Cisco Unified Communications Manager (CUCM) audit events sent by remote syslog:
#   %UC_AUDITLOG-<sev>-AdministrativeEvent (or SecurityEvent, CriticalEvent): %[UserID=...][ClientAddress=...]
#     [Severity=...][EventType=...][ResourceAccessed=...][EventStatus=...][AuditDetails=...][ComponentID=...]
#     [CompulsoryEvent=...][AuditCategory=...][AppID=...][ClusterID=...][NodeID=...]: Audit Event is generated by this application
#   and the form the CUCM Administration Guide shows:
#   CCM_TOMCAT-GENERIC-<sev>-AuditEventGenerated: Audit Event Generated UserID:... Client IP Address:... Severity:...
#     EventType:... ResourceAccessed: ... EventStatus:... Description: ... App ID:... Cluster ID:... Node ID:...
# Child of the global parser Cisco_IOS_Master, which claims CUCM's lines and leaves them without a
# schema. In Sophos Fusion, turn on Set Global Parent in Parser Script. The parser reads the whole
# line itself, not the parent's variables.
# Every audit event except the UserLogging type (sign-ins and sign-outs, which go to Auth) to
# ManagementEvent: configuration changes (GeneralConfigurationUpdate), user and role changes,
# access checks (UserAccess), CLI commands (CLICommand), service starts and stops, and the other
# events of the CUCM audit log. The confirm string spells out "any EventType but UserLogging",
# since the pattern can't say "not".

line = originalData$
hdr = line|(?P<mon>[A-Z][a-z]{2})\s+(?P<day>\d{1,2})\s+(?:(?P<yr>\d{4})\s+)?(?P<hms>\d{1,2}:\d{2}:\d{2})(?:\s+(?P<ap>[AP]M))?(?:\.(?P<frac>\d+))?(?:\s+(?P<tz>[A-Za-z]{2,5}))?\s*:\s+%?(?P<fac>UC_AUDITLOG|CCM_TOMCAT-GENERIC)-(?P<sev>\d)-(?P<mn>[A-Za-z0-9_]+)\s?:
host = line|^(?:<\d{1,3}>)?\s*\d+:\s+(?P<h>[A-Za-z0-9][A-Za-z0-9._-]*):\s+[A-Z][a-z]{2}\s

sensorType$ = "Cisco Unified CM"
sensorEventId$ = STRING(hdr["fac"]) + "-" + STRING(hdr["sev"]) + "-" + STRING(hdr["mn"])

# The time, read as UTC, which CUCM writes. A time in another zone keeps the ingest time. A time
# without the year is read to the second.
zone = UPPERCASE(STRING(hdr["tz"]))
utc = zone == "" OR zone == "UTC" OR zone == "GMT"
frac = IF hdr["frac"] != "" THEN "." + hdr["frac"] ELSE ""
ap = IF hdr["ap"] != "" THEN " " + hdr["ap"] ELSE ""
clock = STRING(hdr["mon"]) + " " + STRING(hdr["day"]) + " " + STRING(hdr["yr"]) + " " + STRING(hdr["hms"]) + frac + ap
short = STRING(hdr["mon"]) + " " + STRING(hdr["day"]) + " " + STRING(hdr["hms"]) + frac + ap
layout = IF hdr["ap"] != "" THEN "03:04:05 PM" ELSE "15:04:05"
eventTimeUsec$ = IF utc AND hdr["yr"] != "" THEN DATETIME(clock, "Jan _2 2006 " + layout) ELSE eventTimeUsec$
eventTimeUsec$ = IF utc AND hdr["yr"] == "" AND hdr["hms"] != "" THEN DATETIME(short, "Jan _2 " + layout, TRUE) ELSE eventTimeUsec$

# The bracketed values, with or without spaces around the key.
kUser = line|\[\s*UserID\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kClient = line|\[\s*ClientAddress\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kType = line|\[\s*EventType\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kStatus = line|\[\s*EventStatus\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kDetails = line|\[\s*AuditDetails\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kComp = line|\[\s*ComponentID\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kCluster = line|\[\s*Cluster\s?ID\s*[=]\s*(?P<v>[^\]]*?)\s*\]
kNode = line|\[\s*Node\s?ID\s*[=]\s*(?P<v>[^\]]*?)\s*\]
# The Administration Guide's form, label by label.
leg = line|AuditEventGenerated\s?:\s*Audit Event Generated\s+UserID:\s*(?P<user>\S*)\s+Client IP Address:\s*(?P<ip>\S*)\s+Severity:\s*\S*\s+EventType:\s*(?P<et>\S*)\s+ResourceAccessed:\s*(?P<res>.*?)\s+EventStatus:\s*(?P<st>\S*)\s+Description:\s*(?P<desc>.*?)\s+App ID:\s*.*?\s+Cluster ID:\s*(?P<cl>.*?)\s+Node ID:\s*(?P<node>\S*)

user = STRING(kUser["v"]) + STRING(leg["user"])
username$ = IF user != "" THEN user ELSE NULL
ip = STRING(kClient["v"]) + STRING(leg["ip"])
sourceAddress$ = IF ip != "" THEN ip ELSE NULL
et = STRING(kType["v"]) + STRING(leg["et"])
operationName$ = IF et != "" THEN et ELSE NULL
# AuditDetails (Description in the guide's form) says what changed, such as
# "record in table device, with key field name = SEP00005E005341 deleted".
details = STRING(kDetails["v"]) + STRING(leg["desc"])
operation$ = IF details != "" THEN details ELSE NULL
# A CLI command (EventType CLICommand) names the command after "CLI Command->".
cli = details|CLI Command\s?-\s?>\s*(?P<c>.+)$
command$ = IF cli["c"] != "" THEN cli["c"] ELSE NULL
st = STRING(kStatus["v"]) + STRING(leg["st"])
result$ = IF st != "" THEN st ELSE NULL
# The component that logged it, such as Cisco CUCM Administration. ResourceAccessed (such as
# CUCMAdmin) has no field here and stays in original_data.
comp = STRING(kComp["v"])
channel$ = IF comp != "" THEN comp ELSE NULL
cluster = STRING(kCluster["v"]) + STRING(leg["cl"])
namespace$ = IF cluster != "" THEN cluster ELSE NULL
node = STRING(kNode["v"]) + STRING(leg["node"])
hname = STRING(host["h"])
hostId$ = IF node != "" THEN node ELSE NULL
hostId$ = IF hostId$ == NULL AND hname != "" THEN hname ELSE hostId$
targetHostname$ = hostId$
