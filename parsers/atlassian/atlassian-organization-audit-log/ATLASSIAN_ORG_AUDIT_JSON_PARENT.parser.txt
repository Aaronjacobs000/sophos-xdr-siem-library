# Source: Atlassian organization audit log events from the Organizations REST API events stream,
# pulled by the example collector in collectors/atlassian-org-audit and posted to HTTP Ingest as
# JSON lines. The collector puts "atlassian_record": "event" first and adds orgId, and the action's
# display name and group from the event actions list (actionDisplayName, actionGroup).
# Parent: hands sign-ins and sign-outs to the Auth child, and every other event to the CloudAudit
# child. Atlassian doesn't publish its action keys, so the routing reads the display names from
# Atlassian's audit log activities list, and the one key that is public, user_login.
# !TRIMALLOFF stops TRIM_ALL() stripping braces and brackets from JSON values.

json = JSON(originalData$)

# Variables and $ fields set here are visible to both children.
sensorType$ = "Atlassian"
act = json["$.attributes.action"]
display = json["$.actionDisplayName"]
# time is ISO 8601 in UTC, with milliseconds.
eventTimeUsec$ = DATETIME(json["$.attributes.time"], "2006-01-02T15:04:05.999999999Z07:00")

authEvent = act == "user_login" OR display == "Logged in to account" OR display == "Logged out of account"
eventType$ = IF authEvent THEN "auth" ELSE "audit"
