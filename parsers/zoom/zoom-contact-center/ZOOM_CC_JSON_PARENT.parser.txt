# Source: Zoom Contact Center operation logs (GET /contact_center/reports/operation_logs) and voice
# call logs (GET /contact_center/voice_calls), pulled by the example bridge in
# bridges/zoom-report-logs and posted to HTTP Ingest as JSON lines. The bridge puts "zoom_record"
# first ("cc_operation" or "cc_voice_call") and adds account_id.
# Parent: hands operation logs to the CloudAudit child. Voice calls fit no schema, so they stay here
# in Generic, with the engagement ID as the event ID and the call's start as the event time.
# !TRIMALLOFF stops TRIM_ALL() stripping braces and brackets from JSON values.

json = JSON(originalData$)

# Variables and $ fields set here are visible to the child.
sensorType$ = "Zoom Contact Center"
kind = json["$.zoom_record"]
call = kind == "cc_voice_call"
# Operation log times end in Z; the bridge asks for voice calls in UTC, so their times end in +00:00.
when = IF call THEN json["$.start_time"] ELSE json["$.time"]
eventTimeUsec$ = DATETIME(when, "2006-01-02T15:04:05.999999999Z07:00")
sensorEventId$ = IF call THEN json["$.engagement_id"] ELSE json["$.operation_log_id"]
eventType$ = IF call THEN "call" ELSE "audit"
