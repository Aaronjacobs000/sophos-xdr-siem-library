# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Zoom's sign-in and sign-out activity report (GET /report/activities) from the example
# bridge. The report lists completed sign-ins and sign-outs only; failed sign-ins aren't in it.
# Child of ZOOM_REPORTS_JSON_PARENT: each one to Auth, with the user's email as the target user
# and the address they signed in from.
# json, kind, eventType$, sensorType$ and eventTimeUsec$ come from the parent.

authSystem$ = "Zoom"
targetUserName$ = json["$.email"]
sourceAddress$ = json["$.ip_address"]
kindOf = json["$.type"]
status$ = kindOf
logout = kindOf == "Sign out"
authCategory$ = IF logout THEN "ACCOUNT_LOGOFF" ELSE "ACCOUNT_LOGON"
action$ = IF logout THEN "LOGOFF" ELSE "LOGON"
actionResult$ = "SUCCESS"
# client_type is the Zoom client used, such as Browser, Windows, Mac, iOS or Android.
logonApplicationFamily$ = json["$.client_type"]

m1.key$ = "account_id"
m1.value$ = json["$.account_id"]
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [m1]
m2.key$ = "client_version"
m2.value$ = json["$.version"]
eventMetadata$.record$ = IF m2.value$ THEN eventMetadata$.record$ + [m2] ELSE eventMetadata$.record$
