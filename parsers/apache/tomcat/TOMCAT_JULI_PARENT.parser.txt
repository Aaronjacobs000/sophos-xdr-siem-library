!SAMPLE=20-Oct-2026 22:20:11.010 INFO [main] org.apache.catalina.startup.Catalina.start Server startup in [1534] milliseconds
!CONFIRMWITH=PATTERN
!CONFIRMSTRING=\d{2}-[A-Z][a-z]{2}-\d{4} \d{2}:\d{2}:\d{2}(?:\.\d{1,3})? (?:SEVERE|WARNING|INFO|CONFIG|FINE|FINER|FINEST) \[[^\]]*\] org\.apache\.(?:catalina|coyote|tomcat|jasper|juli)\.
!SCHEMA=scwx.generic

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Apache Tomcat's own logs (catalina, localhost and manager logs, and catalina.out) in the
# default OneLineFormatter layout, read from the files by a forwarder and sent to the data collector:
#   <dd-Mon-yyyy HH:mm:ss.SSS> <LEVEL> [<thread>] <class>.<method> <message>
# Parent: claims lines from Tomcat's own classes (org.apache.catalina, coyote, tomcat, jasper and
# juli) and hands them on:
#   LockOutRealm: An attempt was made to authenticate the locked user [<user>]     to the Auth child
#   HostConfig deploying and undeploying, Manager deploy, undeploy, start, stop, restart, install
#   and WAR uploads, and shutdown commands on the shutdown port                    to the Management Event child
# Other lines stay generic. A web application's own lines don't match, and neither do lines of a
# stack trace after the first.

l = originalData$|(?P<t>\d{2}-[A-Z][a-z]{2}-\d{4} \d{2}:\d{2}:\d{2})(?:\.\d{1,3})? (?P<level>SEVERE|WARNING|INFO|CONFIG|FINE|FINER|FINEST) \[(?P<thread>[^\]]*)\] (?P<src>org\.apache\.(?:catalina|coyote|tomcat|jasper|juli)\.\S+) (?P<msg>.*?)\s*$
hdr = originalData$|^(?:<\d{1,3}>)?(?:[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2}|\d{4}-\d{2}-\d{2}T\S+)\s+(?P<host>\S+)\s+(?P<tag>[^\s:\[]+)(?:\[\d+\])?:\s
src = STRING(l["src"])
msg = STRING(l["msg"])
level = STRING(l["level"])

# Variables and $ fields set here are visible to every child.
sensorType$ = "Apache Tomcat"
host = STRING(hdr["host"])
hostId$ = IF host != "" THEN host ELSE NULL
# The time has no zone: it is the JVM's local time, read as UTC. Run Tomcat with
# -Duser.timezone=UTC, or the event times are off by the offset.
eventTimeUsec$ = DATETIME(l["t"], "02-Jan-2006 15:04:05")

# The Manager application logs through its servlet: "<Manager|HTMLManager>: <command>: <text>".
mg = msg|^(?P<servlet>(?:HTML)?Manager):\s+(?:(?P<cmd>deploy|undeploy|start|stop|restart|install):\s+)?(?P<text>.*)$
servlet = STRING(mg["servlet"])
mcmd = STRING(mg["cmd"])
upload = servlet != "" AND CONTAINS(STRING(mg["text"]), "Uploading WAR file to ")
hostDeploy = CONTAINS(src, ".HostConfig.") AND (CONTAINS(msg, "Deploying ") OR CONTAINS(msg, "Undeploying ") OR CONTAINS(msg, "Reloading context "))
shutdown = CONTAINS(src, ".StandardServer.") AND (CONTAINS(msg, "shutdown command"))
eventType$ = "other"
eventType$ = IF CONTAINS(src, ".LockOutRealm.") AND CONTAINS(msg, "authenticate the locked user") THEN "lockout" ELSE eventType$
eventType$ = IF (servlet != "" AND mcmd != "") OR upload OR hostDeploy OR shutdown THEN "admin" ELSE eventType$
