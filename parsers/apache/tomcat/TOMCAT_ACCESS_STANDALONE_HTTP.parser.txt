# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Apache Tomcat access logs, read from the log files by a forwarder (Tomcat has no syslog
# output) and sent to the data collector:
#   AccessLogValve with the default pattern (common, %h %l %u %t "%r" %s %b):
#     <client> <ident> <user> [<time>] "<request>" <status> <bytes>
#   JsonAccessLogValve (Tomcat 9.0.74 and 10.1.8 or later) with a pattern that has %t before %s:
#     {"host":"<client>","logicalUserName":"-","user":"-","time":"[<time>]","request":"<request>","statusCode":"<status>","size":"<bytes>"}
# Standalone parser for the HTTP schema. The combined pattern (with referer and user agent) in the
# AccessLogValve is read by the library's nginx and Apache access log parser instead.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

hdr = originalData$|^(?:<\d{1,3}>)?(?:[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2}|\d{4}-\d{2}-\d{2}T\S+)\s+(?P<host>\S+)\s+(?P<tag>[^\s:\[]+)(?:\[\d+\])?:\s
jm = originalData$|(?P<j>\{"[^"]+":.*\})\s*$
isJson = STRING(jm["j"]) != ""
j = IF isJson THEN JSON(jm["j"]) ELSE NULL
cm = originalData$|(?:^|\s)(?P<client>\S+)\s+\S+\s+(?P<user>\S+)\s+\[(?P<time>\d{2}/[A-Z][a-z]{2}/\d{4}:\d{2}:\d{2}:\d{2}\s[+-]\d{4})\]\s+"(?P<request>(?:[^"\\]|\\.)*)"\s+(?P<status>\d{3})\s+(?P<bytes>\d+|-)\s*$

sensorType$ = "Apache Tomcat"
host = STRING(hdr["host"])
hostId$ = IF host != "" THEN host ELSE NULL

client = IF isJson THEN STRING(j["$.host"]) ELSE STRING(cm["client"])
user = IF isJson THEN STRING(j["$.user"]) ELSE STRING(cm["user"])
# %t keeps its brackets in the JSON valve.
tj = IF isJson THEN STRING(j["$.time"]) ELSE ""
tjm = tj|^\[(?P<t>[^\]]+)\]$
time = IF isJson THEN STRING(tjm["t"]) ELSE STRING(cm["time"])
request = IF isJson THEN STRING(j["$.request"]) ELSE STRING(cm["request"])
status = IF isJson THEN STRING(j["$.statusCode"]) ELSE STRING(cm["status"])
size = IF isJson THEN STRING(j["$.size"]) ELSE STRING(cm["bytes"])

# The request time has its own offset, so it doesn't depend on the forwarder's header.
eventTimeUsec$ = IF time != "" THEN DATETIME(time, "02/Jan/2006:15:04:05 -0700") ELSE eventTimeUsec$
# %h is the client address, or its name when enableLookups is on.
sourceAddress$ = IF IS_VALID_IP(client) THEN client ELSE NULL
sourceUsername$ = IF user != "" AND user != "-" THEN user ELSE NULL
responseCode$ = status
# %b is the bytes sent to the client, - for none.
rxByteCount$ = IF size != "-" THEN size ELSE NULL
requestDirection$ = "INBOUND"
ua = IF isJson THEN STRING(j["$.requestHeaders['User-Agent']"]) ELSE ""
userAgent$ = IF ua != "" AND ua != "-" THEN ua ELSE NULL
ref = IF isJson THEN STRING(j["$.requestHeaders.Referer"]) ELSE ""
referer$ = IF ref != "" AND ref != "-" THEN ref ELSE NULL

# A malformed request has no method.
req = request|^(?P<method>[A-Z]+)\s+(?P<path>[^?\s]*)(?:\?(?P<query>\S*))?(?:\s+(?P<version>HTTP/[\d.]+))?$
method = STRING(req["method"])
httpMethod$ = IF method != "" THEN method ELSE NULL
path = STRING(req["path"])
query = STRING(req["query"])
# The log keeps the path and query as the client sent them, still percent-encoded.
uriPathRaw$ = IF path != "" THEN path ELSE NULL
uriPath$ = IF path != "" THEN path ELSE NULL
uriQueryRaw$ = IF query != "" THEN query ELSE NULL
uriQuery$ = IF query != "" THEN query ELSE NULL

r1.key$ = "log_format"
r1.value$ = IF isJson THEN "json" ELSE "common"
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [r1]
r2.key$ = "syslog_tag"
r2.value$ = STRING(hdr["tag"])
eventMetadata$.record$ = IF r2.value$ != "" THEN eventMetadata$.record$ + [r2] ELSE eventMetadata$.record$
r3.key$ = "http_version"
r3.value$ = STRING(req["version"])
eventMetadata$.record$ = IF r3.value$ != "" THEN eventMetadata$.record$ + [r3] ELSE eventMetadata$.record$
r4.key$ = "request"
r4.value$ = request
eventMetadata$.record$ = IF method == "" AND request != "" THEN eventMetadata$.record$ + [r4] ELSE eventMetadata$.record$
r5.key$ = "client"
r5.value$ = client
eventMetadata$.record$ = IF client != "" AND NOT IS_VALID_IP(client) THEN eventMetadata$.record$ + [r5] ELSE eventMetadata$.record$
