# Parent: hands sign-ins, failed sign-ins and sign-outs to the Auth child, Slack's anomaly events
# to the Thirdparty child, and every other entry to the CloudAudit child.
# !TRIMALLOFF stops TRIM_ALL() stripping braces and brackets from JSON values.

# Variables and $ fields set here are visible to every child.
sensorType$ = "Slack Audit Logs"
sensorEventId$ = json["$.id"]
kind = json["$.action"]
etype = json["$.entity.type"]
# date_create is Unix time in seconds.
eventTimeUsec$ = INT(STRING(json["$.date_create"]), 10) * 1000000

# Every sign-out action starts with user_logout, such as user_logout_compromised.
signin = kind == "user_login" OR kind == "user_login_failed" OR kind == "cli_login" OR CONTAINS(kind, "user_logout")
eventType$ = "audit"
eventType$ = IF kind == "anomaly" THEN "alert" ELSE eventType$
