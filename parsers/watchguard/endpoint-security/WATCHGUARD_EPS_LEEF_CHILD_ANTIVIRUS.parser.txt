# v, evType, sevText, hostName, hostIp, sensorType$, hostId$ and eventTimeUsec$ come from the parent.

hash = TRIM(STRING(v["MWHash"]))

threatType = TRIM(STRING(v["ThreatType"]))
threatName$ = IF name != "" THEN name ELSE threatType
filePath$ = IF path != "" THEN path ELSE NULL
