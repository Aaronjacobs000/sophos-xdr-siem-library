# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: nginx or Apache httpd access logs in the combined format, over syslog to the data collector.
# Standalone parser for the HTTP schema. The two servers write the same combined format:
#   <client> <ident> <user> [<time>] "<request>" <status> <bytes> "<referer>" "<user agent>"
# nginx: access_log syslog:server=<collector> combined; (tag nginx).
# Apache: CustomLog "|/usr/bin/logger -t httpd -p local6.info" combined, forwarded by the host.

hdr = originalData$|^(?:<\d{1,3}>)?(?:[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2}|\d{4}-\d{2}-\d{2}T\S+)\s+(?P<host>\S+)\s+(?P<tag>[^\s:\[]+)(?:\[\d+\])?:\s
m = originalData$|:\s+(?P<client>\S+)\s+\S+\s+(?P<user>\S+)\s+\[(?P<time>\d{2}/[A-Z][a-z]{2}/\d{4}:\d{2}:\d{2}:\d{2}\s[+-]\d{4})\]\s+"(?P<request>(?:[^"\\]|\\.)*)"\s+(?P<status>\d{3})\s+(?P<bytes>\d+|-)\s+"(?P<referer>(?:[^"\\]|\\.)*)"\s+"(?P<agent>(?:[^"\\]|\\.)*)"\s*$

tag = LOWERCASE(hdr["tag"])
sensorType$ = "Web server"
sensorType$ = IF CONTAINS(tag, "nginx") THEN "nginx" ELSE sensorType$
sensorType$ = IF CONTAINS(tag, "httpd") OR CONTAINS(tag, "apache") THEN "Apache httpd" ELSE sensorType$
hostId$ = hdr["host"]

# The request time has its own offset, so it doesn't depend on the syslog header.
eventTimeUsec$ = DATETIME(m["time"], "02/Jan/2006:15:04:05 -0700")

sourceAddress$ = m["client"]
# "-" means no user; SANITIZE_ALL turns it into NULL. Apache also logs "-" for zero bytes.
sourceUsername$ = m["user"]
responseCode$ = INT(m["status"], 10)
rxByteCount$ = m["bytes"]
referer$ = m["referer"]
userAgent$ = m["agent"]
requestDirection$ = "INBOUND"

# A malformed request (a TLS handshake sent to a plain HTTP port, for example) has no method.
req = m["request"]|^(?P<method>[A-Z]+)\s+(?P<path>[^?\s]*)(?:\?(?P<query>\S*))?(?:\s+(?P<version>HTTP/[\d.]+))?$
httpMethod$ = req["method"]
# The log keeps the path and query as the client sent them, still percent-encoded.
uriPathRaw$ = req["path"]
uriPath$ = req["path"]
uriQueryRaw$ = req["query"]
uriQuery$ = req["query"]

r1.key$ = "syslog_tag"
r1.value$ = hdr["tag"]
eventMetadata$.record$ = [r1]
r2.key$ = "http_version"
r2.value$ = req["version"]
eventMetadata$.record$ = IF req["version"] != "" THEN eventMetadata$.record$ + [r2] ELSE eventMetadata$.record$
r3.key$ = "request"
r3.value$ = m["request"]
eventMetadata$.record$ = IF req["method"] == "" THEN eventMetadata$.record$ + [r3] ELSE eventMetadata$.record$
