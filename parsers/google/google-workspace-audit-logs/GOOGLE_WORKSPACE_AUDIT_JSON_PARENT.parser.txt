# Source: Google Workspace audit activities from the Admin SDK Reports API, pulled by the example
# bridge in bridges/google-workspace-audit and posted to HTTP Ingest as JSON lines. The bridge
# sends one record per event: the activity with "google_workspace_record": "activity" first and
# only that event in events, plus params, the event's parameters keyed by name.
# Parent: hands sign-ins, sign-outs and sign-in challenges (login and SAML) and ChromeOS sign-ins
# to the Auth child; Google's account warnings and Chrome's malware, unsafe site, password and
# data loss events to the Thirdparty child; and every other event (admin console changes,

json = JSON(originalData$)

# Variables and $ fields set here are visible to every child.
sensorType$ = "Google Workspace"
app = json["$.id.applicationName"]
evName = json["$.events[0].name"]
evType = json["$.events[0].type"]
# id.time is RFC 3339 in UTC, with milliseconds.
eventTimeUsec$ = DATETIME(json["$.id.time"], "2006-01-02T15:04:05.999999999Z07:00")

loginAuth = app == "login" AND (evName == "login_success" OR evName == "login_failure" OR evName == "logout" OR evName == "login_challenge" OR evName == "login_verification")
chromeAuth = app == "chrome" AND (evName == "CHROME_OS_LOGIN_EVENT" OR evName == "CHROME_OS_LOGIN_FAILURE_EVENT" OR evName == "CHROME_OS_LOGOUT_EVENT")
authEvent = loginAuth OR app == "saml" OR chromeAuth

loginAlert = app == "login" AND (evType == "account_warning" OR evType == "attack_warning")
chromeAlert = app == "chrome" AND (evName == "MALWARE_TRANSFER" OR evName == "UNSAFE_SITE_VISIT" OR evName == "PASSWORD_REUSE" OR evName == "PASSWORD_BREACH" OR evName == "SENSITIVE_DATA_TRANSFER" OR evName == "DLP_EVENT")
alertEvent = loginAlert OR chromeAlert

eventType$ = "audit"
eventType$ = IF alertEvent THEN "alert" ELSE eventType$
eventType$ = IF authEvent THEN "auth" ELSE eventType$
