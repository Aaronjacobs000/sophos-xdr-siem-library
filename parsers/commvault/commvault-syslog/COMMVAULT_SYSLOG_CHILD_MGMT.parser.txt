# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Commvault CommServe syslog forwarding of events, the audit trail and alerts.
# Child of COMMVAULT_SYSLOG_PARENT: audit trail records other than logins, and events other
# than anomaly detections, to Management Event.
# kind, text, user, operation, program, computer, commcell, recordId, jobId, cvSeverity,
# eventType$, sensorType$, hostId$ and eventTimeUsec$ come from the parent.

audit = kind == "audit"
channel$ = IF audit THEN "Commvault audit trail" ELSE "Commvault events"
# An audit record names the operation; an event names the program that raised it.
operationName$ = IF audit THEN operation ELSE program
operation$ = IF text != "" THEN text ELSE NULL
username$ = IF user != "" THEN user ELSE NULL
targetHostname$ = IF computer != "" THEN computer ELSE commcell
# The record's severity (Information, Minor, Major or Critical for events; Low, Medium, High or
# Critical for the audit trail) is the only outcome Commvault gives.
result$ = IF cvSeverity != "" THEN cvSeverity ELSE NULL
rid = recordId|^(?P<n>\d+)$
eventId$ = IF rid["n"] != "" THEN INT(rid["n"], 10) ELSE NULL
# The job the record belongs to, when there is one.
job = jobId|^(?P<n>\d+)$
operationId$ = IF job["n"] != "" THEN INT(job["n"], 10) ELSE NULL
