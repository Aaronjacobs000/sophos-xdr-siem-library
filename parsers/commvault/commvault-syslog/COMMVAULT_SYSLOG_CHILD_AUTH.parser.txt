# The account is in Username, or in the details text in one of these forms:
#   username:[DOMAIN\user]   Login Name: DOMAIN\user   for user [DOMAIN\user]
d1 = text|username:\[(?P<u>[^\]]*)\]
d2 = text|Login Name:\s*(?P<u>\S+)
d3 = text|for user \[(?P<u>[^\]]*)\]
acct = user
acct = IF d3["u"] != "" THEN d3["u"] ELSE acct
acct = IF d2["u"] != "" THEN d2["u"] ELSE acct
acct = IF d1["u"] != "" THEN d1["u"] ELSE acct
dom = acct|^(?P<d>[^\\]+)\\(?P<n>.+)$
targetUserName$ = IF dom["n"] != "" THEN dom["n"] ELSE acct
targetDomainName$ = IF dom["d"] != "" THEN dom["d"] ELSE NULL
targetHostName$ = commcell
authSystem$ = "Commvault CommServe"

src = text|host/\s*device:\[(?P<ip>[^\]]*)\]
# IS_VALID_IP() fails the event on NULL, which a capture that matched nothing gives.

app = text|(?:logged in|login attempted) from:\s*\[(?P<a>[^\]]*)\]
logonApplicationFamily$ = IF app["a"] != "" THEN app["a"] ELSE NULL

op = LOWERCASE(operation)
failed = CONTAINS(op, "login fail")
locked = CONTAINS(op, "account locked")
loggedOut = CONTAINS(op, "logged out")

action$ = IF failed THEN "FAILURE" ELSE action$
action$ = IF locked THEN "ACCOUNT" ELSE action$

authCategory$ = IF locked THEN "ACCOUNT_LOCK" ELSE authCategory$
