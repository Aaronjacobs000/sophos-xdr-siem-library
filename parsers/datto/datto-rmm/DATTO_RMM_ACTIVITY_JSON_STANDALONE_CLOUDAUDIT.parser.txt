# Standalone parser for CloudAudit: every activity (logins, jobs and components run, remote
# takeover and Web Remote sessions, file transfers, device and user changes), with the event
# name built from category and action. Fields follow the ActivityLog schema in the Datto RMM
# API spec; Datto doesn't publish the list of categories and actions.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

json = JSON(originalData$)

sensorType$ = "Datto RMM"
eventSource$ = "Datto RMM"
category = json["$.category"]
actionName = json["$.action"]
eventName$ = category + "." + actionName
eventType$ = json["$.entity"]
managementEvent$ = TRUE
readOnly$ = FALSE

# date is seconds since 1970 as a decimal number; STRING() drops the fraction.
eventTimeUsec$ = INT(STRING(json["$.date"]), 10) * 1000000

userName$ = json["$.user.userName"]
userId$ = STRING(json["$.user.id"])
host = json["$.hostname"]
targetHostname$ = host
hostId$ = host

d1.key$ = "activity_id"
d1.value$ = json["$.id"]
d2.key$ = "site_name"
d2.value$ = json["$.site.name"]
# Assign the list first: reading additionalEventData$.record$ before it is set fails the event.
additionalEventData$.record$ = [d1, d2]
d3.key$ = "details"
d3.value$ = json["$.details"]
additionalEventData$.record$ = IF d3.value$ THEN additionalEventData$.record$ + [d3] ELSE additionalEventData$.record$
d4.key$ = "device_id"
d4.value$ = STRING(json["$.deviceId"])
additionalEventData$.record$ = IF d4.value$ THEN additionalEventData$.record$ + [d4] ELSE additionalEventData$.record$
