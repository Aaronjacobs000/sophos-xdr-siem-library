agentVersion$ = IF version != "" THEN version ELSE NULL

objectName = STRING(v["ObjectName"])
path = STRING(v["filePath"])
path = IF path == "" THEN objectName ELSE path
web = path|^(?P<u>https?://.+)$
urlString$ = IF web["u"] != "" THEN web["u"] ELSE NULL
filePath$ = IF web["u"] == "" AND path != "" THEN path ELSE NULL
threatType$ = IF web["u"] == "" AND path != "" THEN "FILE" ELSE "UNKNOWN_THREATTYPE"

# Kaspersky documents GNRL_EA_PARAM_1 of GNRL_EV_VIRUS_FOUND as cs4 with the label SHA256.
sha = STRING(v["SHA256"])
shaHex = sha|^(?P<h>[0-9A-Fa-f]{64})$
fileHash$.sha256$ = IF shaHex["h"] != "" THEN LOWERCASE(shaHex["h"]) ELSE NULL

# The event type says what happened to the object.
actionTaken$ = "UNKNOWN_ACTIONTAKEN"
actionTaken$ = IF eventId == "GNRL_EV_VIRUS_FOUND" OR eventId == "GNRL_EV_VIRUS_FOUND_BY_KSN" OR eventId == "GNRL_EV_SUSPICIOUS_OBJECT_FOUND" OR eventId == "GNRL_EV_OBJECT_REPORTED" THEN "ALERT" ELSE actionTaken$
actionTaken$ = IF eventId == "GNRL_EV_OBJECT_DELETED" THEN "DELETED" ELSE actionTaken$
actionTaken$ = IF eventId == "GNRL_EV_OBJECT_QUARANTINED" THEN "QUARANTINED" ELSE actionTaken$
actionTaken$ = IF eventId == "GNRL_EV_OBJECT_BLOCKED" OR eventId == "GNRL_EV_VIRUS_FOUND_AND_BLOCKED" THEN "BLOCKED" ELSE actionTaken$

# DetectionType is Kaspersky's threat type: 60 virus, 61 trojan, 62 malware, 63 backdoor,
# 64 worm, 65 riskware, 66 adware, 67 pornware, 69 behavioral, 70 phishing, 71 malicious link.
# Names that start with not-a-virus are legitimate software that can be misused.
dt = STRING(v["DetectionType"])
threatCategory$ = "UNKNOWN_THREATCATEGORY"
threatCategory$ = IF dt == "60" THEN "VIRUS" ELSE threatCategory$
threatCategory$ = IF dt == "61" THEN "TROJAN" ELSE threatCategory$
threatCategory$ = IF dt == "62" OR dt == "64" THEN "MALWARE" ELSE threatCategory$
threatCategory$ = IF dt == "63" THEN "BACKDOOR" ELSE threatCategory$
threatCategory$ = IF dt == "65" OR dt == "66" OR dt == "67" THEN "APPLICATION" ELSE threatCategory$
threatCategory$ = IF CONTAINS(LOWERCASE(threatName$), "ransom") THEN "RANSOMWARE" ELSE threatCategory$
threatCategory$ = IF CONTAINS(LOWERCASE(threatName$), "not-a-virus") THEN "APPLICATION" ELSE threatCategory$
