!CONFIRMWITH=PATTERN
!CONFIRMSTRING=\[FILTER\]\[(?:Block|Pass)\]\[[^\]]*->[^\]]*,\s*[\d:]+\s*\]\[@S:R=\d+:\d+,\s*[^\]]*->[^\]]*\]\[[^\]]+\]

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: DrayTek Vigor routers (DrayOS), firewall IP Filter syslog (local0) to the data collector.
# Standalone parser for the Netflow schema. Tick Syslog on the filter rules you want logged.
# Format, from DrayTek's "Interpreting Firewall and CSM logs" guide:
#   [FILTER][<Block|Pass>][<from>-><to>, <uptime> ][@S:R=<set>:<rule>, <src>[:<port>]-><dst>[:<port>]][<protocol>][<packet details>]
# Setting source_address and destination_address lets the IP Watchlist detector use these events.

m = originalData$|^(?:<\d{1,3}>)?(?:(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+)?(?:(?P<host>[^\s\[]+?):?\s+)?\[FILTER\]\[(?P<action>Block|Pass)\]\[(?P<dir>[^,\]]+),\s*(?P<uptime>[\d:]+)\s*\]\[@S:R=(?P<set>\d+):(?P<rule>\d+),\s*(?P<src>\d{1,3}(?:\.\d{1,3}){3})(?::(?P<sport>\d+))?->(?P<dst>\d{1,3}(?:\.\d{1,3}){3})(?::(?P<dport>\d+))?\]\[(?P<proto>[^\]]+)\]\[(?P<info>[^\]]*)\]

sensorType$ = "DrayTek Vigor"
hostId$ = m["host"]
# The BSD syslog time is the router's clock, with no year and no zone: DATETIME picks the
# year (TRUE) and reads UTC.
eventTimeUsec$ = DATETIME(m["time"], "Jan _2 15:04:05", TRUE)

sourceAddress$ = m["src"]
destinationAddress$ = m["dst"]
sourcePort$ = m["sport"]
destinationPort$ = m["dport"]

proto = UPPERCASE(m["proto"])
protocol$ = NULL
protocol$ = IF proto == "TCP" THEN 6 ELSE protocol$
protocol$ = IF proto == "UDP" THEN 17 ELSE protocol$
protocol$ = IF proto == "ICMP" THEN 1 ELSE protocol$
protocol$ = IF proto == "GRE" THEN 47 ELSE protocol$

flowAction$ = IF m["action"] == "Block" THEN "FW_BLOCKED" ELSE "FW_ALLOWED"

# WAN->LAN/RT is traffic from the Internet; LAN/RT/VPN->WAN is outgoing. LAN to LAN has no direction.
dir = m["dir"]
inbound = dir|^WAN->
outbound = dir|->WAN$
direction$ = NULL
direction$ = IF inbound[0] != "" THEN "INBOUND" ELSE direction$
direction$ = IF outbound[0] != "" THEN "OUTBOUND" ELSE direction$

# Set 13 is the Default Rule in Firewall > General Setup.
r1.key$ = "filter_set"
r1.value$ = m["set"]
r2.key$ = "filter_rule"
r2.value$ = m["rule"]
r3.key$ = "filter_direction"
r3.value$ = dir
r4.key$ = "router_uptime"
r4.value$ = m["uptime"]
r5.key$ = "packet_info"
r5.value$ = m["info"]
eventMetadata$.record$ = [r1, r2, r3, r4, r5]
