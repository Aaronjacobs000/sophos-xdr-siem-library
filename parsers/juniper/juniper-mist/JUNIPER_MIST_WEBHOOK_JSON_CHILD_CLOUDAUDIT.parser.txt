# Source: Juniper Mist audits webhooks: every configuration change made in the Mist portal or API,
# with the admin, their address and browser, and the settings before and after.
# Child of JUNIPER_MIST_WEBHOOK_JSON_PARENT: each one to CloudAudit, with the message (such as
# Update Site Settings) as the event name. admin_name is the admin's name and email in one text;
# the email becomes the user name. The site and the device become resources.
# json, topic, ts, eventType$, sensorType$ and eventTimeUsec$ come from the parent.

eventSource$ = "Juniper Mist"
eventName$ = json["$.events[0].message"]
eventType$ = "audit"
org = json["$.events[0].org_id"]
recipientAccountId$ = org

mail = admin|(?P<m>[^\s]+@[^\s]+)\s*$
userName$ = IF mail["m"] THEN mail["m"] ELSE admin
userId$ = json["$.events[0].admin_id"]
sourceAddress$ = json["$.events[0].src_ip"]
userAgent$ = json["$.events[0].user_agent"]
managementEvent$ = TRUE
readOnly$ = FALSE

r1.resourceId$ = site
r1.resourceType$ = "site"
r1.resourceAccountId$ = org
device = json["$.events[0].device_id"]
r2.resourceId$ = device
r2.resourceType$ = "device"
r2.resourceAccountId$ = org
resources$ = []
resources$ = IF site THEN resources$ + [r1] ELSE resources$

d1.value$ = admin
d2.key$ = "site_name"

d4.key$ = "after"

additionalEventData$.record$ = IF d2.value$ THEN additionalEventData$.record$ + [d2] ELSE additionalEventData$.record$
