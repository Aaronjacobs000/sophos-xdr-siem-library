# Source: Juniper Mist webhooks for the audits, alarms and nac-events topics,
# {"topic": ..., "events": [...]}, through the webhook relay in bridges/webhook-relay with
# BRIDGE_SPLIT_BATCHES=1, which forwards each event of a batch as its own record. These parsers
# read the first event of a record, so a batch that wasn't split loses the rest.
# Parent: hands audits to the CloudAudit child, alarms to the Thirdparty child and Mist Access
# Assurance NAC events to the Auth child. Other topics aren't claimed.
# !TRIMALLOFF stops TRIM_ALL() stripping braces and brackets from JSON values.

# Variables and $ fields set here are visible to every child.
sensorType$ = "Juniper Mist"
topic = json["$.topic"]
# timestamp is epoch seconds, with a fraction on audits and NAC events. Some alarm types send
# When instead.
ts = json["$.events[0].timestamp"]
ts = IF ts THEN ts ELSE json["$.events[0].When"]
eventTimeUsec$ = INT(STRING(ts * 1000000), 10)
eventType$ = topic
