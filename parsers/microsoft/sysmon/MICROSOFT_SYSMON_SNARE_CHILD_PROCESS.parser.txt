!CONFIRMWITH=PATTERN
!CONFIRMSTRING=MSWinEventLog\t[^\t]*\tMicrosoft-Windows-Sysmon/Operational\t[^\t]*\t[^\t]*\t1\tMicrosoft-Windows-Sysmon\t

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-ngsiem-parser-studio
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Sysmon event 1 (Process Create) from the Microsoft-Windows-Sysmon/Operational log, sent
# in Snare format over syslog by NXLog or the Snare agent, the feed the Sophos Windows Event Log
# integration reads.
# Child of the global parser SNARE_Unsupported, the child of Microsoft_Windows_SNARE_Master that
# takes the Snare events no other built-in parser handles and leaves them in the generic schema.
# Under SNARE_Unsupported it doesn't compete with it: in Run Test a sibling of SNARE_Unsupported
# lost every line after one SNARE_Unsupported took. In Sophos Fusion, turn on Set Global Parent in
# Parser Script. The parser reads the whole line itself, not the parents' variables.
# Snare fields are tab separated: MSWinEventLog, criticality, log name, Snare counter, date and
# time, event ID, source, user, SID type, event type, computer, category, data string, the
# rendered message, and the event log counter. The agents replace the message's line breaks with
# spaces, so each "Label: value" pair ends where the next label starts.

s = originalData$|MSWinEventLog\t(?P<crit>[^\t]*)\t(?P<log>[^\t]*)\t(?P<counter>[^\t]*)\t(?P<dt>[^\t]*)\t(?P<eid>[^\t]*)\t(?P<source>[^\t]*)\t(?P<user>[^\t]*)\t(?P<sidtype>[^\t]*)\t(?P<etype>[^\t]*)\t(?P<computer>[^\t]*)\t(?P<category>[^\t]*)\t(?P<data>[^\t]*)\t(?P<msg>[^\t]*)(?:\t(?P<logcounter>[^\t]*))?
msg = " " + s["msg"] + " "

sensorType$ = "Microsoft Sysmon"
sensorEventId$ = s["logcounter"]
computerName$ = s["computer"]
hostId$ = s["computer"]
os$.os$ = "OS_WINDOWS"

# UtcTime is UTC with milliseconds. Without it, read the Snare date and time, which Sophos asks
# you to send in UTC.
utc = msg|\sUtcTime: (?P<v>\d{4}-\d\d-\d\d \d\d:\d\d:\d\d(?:\.\d+)?)\s
eventTimeUsec$ = IF utc["v"] != "" THEN DATETIME(utc["v"], "2006-01-02 15:04:05.999") ELSE DATETIME(s["dt"], "Mon Jan 02 15:04:05 2006")
processCreateTimeUsec$ = IF utc["v"] != "" THEN eventTimeUsec$ ELSE NULL

# Fields in Sysmon's order: RuleName, UtcTime, ProcessGuid, ProcessId, Image, FileVersion,
# Description, Product, Company, OriginalFileName, CommandLine, CurrentDirectory, User, LogonGuid,
# LogonId, TerminalSessionId, IntegrityLevel, Hashes, ParentProcessGuid, ParentProcessId,
# ParentImage, ParentCommandLine, ParentUser. Releases before schema 4.81 have no ParentUser.
guid = msg|\sProcessGuid: (?P<v>\{[^}\s]*\})
pid = msg|\sProcessId: (?P<v>\d+)\s
image = msg|\sImage: (?P<v>.*?)\s+(?:FileVersion|Description|Product|Company|OriginalFileName|CommandLine):
cmd = msg|\sCommandLine: (?P<v>.*?)\s+CurrentDirectory:
user = msg|\sUser: (?P<v>.*?)\s+(?:LogonGuid|LogonId):
integrity = msg|\sIntegrityLevel: (?P<v>\S+)\s
hashes = msg|\sHashes: (?P<v>\S+)\s
pguid = msg|\sParentProcessGuid: (?P<v>\{[^}\s]*\})
ppid = msg|\sParentProcessId: (?P<v>\d+)\s
pimage = msg|\sParentImage: (?P<v>.*?)\s+ParentCommandLine:

processCorrelationId$ = guid["v"]
processId$ = pid["v"]
imagePath$ = image["v"]
base = image["v"]|(?P<v>[^\\/]+)$
process$ = base["v"]
commandline$ = cmd["v"]
username$ = user["v"]
parentProcessCorrelationId$ = pguid["v"]
parentProcessId$ = ppid["v"]
parentImagePath$ = pimage["v"]

# Hashes holds the algorithms Sysmon is set to log, for example SHA1=...,MD5=...,SHA256=...,IMPHASH=...
md5 = hashes["v"]|(?:^|,)MD5=(?P<v>[0-9A-Fa-f]{32})
sha1 = hashes["v"]|(?:^|,)SHA1=(?P<v>[0-9A-Fa-f]{40})
sha256 = hashes["v"]|(?:^|,)SHA256=(?P<v>[0-9A-Fa-f]{64})
programHash$.md5$ = md5["v"]
programHash$.sha1$ = sha1["v"]
programHash$.sha256$ = sha256["v"]

# High and System integrity mean an elevated or system token.
lvl = integrity["v"]
processIsAdmin$ = IF lvl == "High" OR lvl == "System" THEN TRUE ELSE NULL
processIsAdmin$ = IF lvl == "Medium" OR lvl == "Low" OR lvl == "AppContainer" OR lvl == "Untrusted" THEN FALSE ELSE processIsAdmin$
