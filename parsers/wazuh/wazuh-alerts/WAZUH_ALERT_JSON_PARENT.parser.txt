# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Wazuh 4.x manager alerts, syslog_output with <format>json</format>, to the data collector.
# Parent: recognizes Wazuh JSON alerts and hands them to its children. The manager sends
#   <132>Mmm dd hh:mm:ss <manager> ossec: {<the alert from alerts.json>}
# Single logon successes and failures go to Auth; every other alert goes to the third-party alert schema.
# !TRIMALLOFF stops TRIM_ALL() stripping braces from JSON values.

jsonMatch = originalData$|ossec:\s*(\{.*})\s*$
json = JSON(jsonMatch[1])

# Variables and $ fields set here are visible to both children.
sensorType$ = "Wazuh"
# The alert time is the manager's local time with its offset, for example 2026-10-12T22:40:14.123+0000.
eventTimeUsec$ = DATETIME(json["$.timestamp"], "2006-01-02T15:04:05.000-0700")
hostId$ = json["$.agent.name"]

# rule.groups, as text, for example "syslog","sshd","authentication_failed".
gm = jsonMatch[1]|"rule"\s*:\s*\{.*?"groups"\s*:\s*\[(?P<g>[^\]]*)\]
groups = REPLACE(gm["g"], "\"", "")

# Wazuh marks single logon events with authentication_success or authentication_failed
# (win_authentication_failed on older rules). authentication_failures marks correlated alerts,
# such as brute force and lockouts, which stay alerts.
eventType$ = "alert"
eventType$ = IF CONTAINS(groups, "authentication_success") OR CONTAINS(groups, "authentication_failed") THEN "auth" ELSE eventType$
