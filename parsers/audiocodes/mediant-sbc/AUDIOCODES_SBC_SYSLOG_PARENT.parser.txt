!SAMPLE=<133>[S=3] [BID=5b1035:7]  Activity Log: WEB: Successful login at 192.0.2.10:443. User: Admin. Session: WEB (198.51.100.23) [Time:21-10@16:39:22.736]
!CONFIRMWITH=PATTERN
!CONFIRMSTRING=^(?:<\d{1,3}>)?(?:[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2}\s+\S+\s+)?\[S\s{0,3}=\s{0,3}\d+\]\s*(?:\[(?:BID=[0-9a-fA-F]+:\d+|SID[=:][0-9a-fA-F:]+)\]\s*)?(?:Activity Log:|RAISE-ALARM:|CLEAR-ALARM:|\||\(\s*N\s+\d+\))
!SCHEMA=scwx.generic

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: AudioCodes Mediant SBC (Mediant 500 to 9000 and Mediant VE and CE, release 7.x) syslog to the
# data collector. The device sends the priority and then its own fields, with no time or host name:
#   <PRI>[S=<sequence>] [BID=<serial>:<restarts>]  Activity Log: <action>. User: <user>. Session: <protocol> (<ip>) [Time:dd-mm@hh:mm:ss.mmm]
#   <PRI>[S=<sequence>] [BID=...]  RAISE-ALARM:<name>; Textual Description: <text>; Severity:<severity>; Source:<source>; Unique ID:<n>;
#   <PRI>[S=<sequence>] |CALL_START |SBC |...   (signaling CDRs, in the CDR format table's columns)
# A syslog relay may add a time and host name in front; that form works too.
# Parent: hands sign-ins and sign-outs to the Auth child, every other Activity Log line (parameter and
# table changes, CLI commands, saves and resets) to the ManagementEvent child, and raised alarms to the
# Thirdparty child. Cleared alarms, CDRs and debug lines stay in the generic schema.

hdr = originalData$|^(?:<\d{1,3}>)?(?:(?P<bsd>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?P<host>\S+)\s+)?\[S\s{0,3}=\s{0,3}(?P<seq>\d+)\]\s*(?:\[(?P<tag>BID|SID)[=:](?P<id>[^\]]*)\]\s*)?(?P<msg>.*)$
msg = hdr["msg"]

# Variables and $ fields set here are visible to the children.
sensorType$ = "AudioCodes Mediant SBC"
host = hdr["host"]
hostId$ = IF host THEN host ELSE NULL
# The board ID is the last 6 characters of the device's serial number and its restart count.
board = hdr["id"]

# An Activity Log line: the action, then the management user, and the protocol and address of the
# session. The action can hold a full stop ("... Login Page. Reason: bad credentials"), so the line is
# read from its ". User: " part.
act = msg|^Activity Log:\s*(?P<text>.*?)\.\s+User:\s*(?P<user>.*?)\.\s+Session:\s*(?P<proto>[^\s(]+)\s*\((?P<ip>[^)]*)\)
activity = act["text"]
signin = activity|^(?P<v>(?:WEB: )?Successful (?:user )?login at |User login succeeded$|Unauthorized access attempt to Login Page|(?:WEB: )?User logout$|User logged out$)
alarm = msg|^RAISE-ALARM:\s*(?P<name>[A-Za-z0-9]+);
# acKeepAlive is a heartbeat trap, not an alarm, so it stays generic.
raised = alarm["name"] AND alarm["name"] != "acKeepAlive"
eventType$ = "other"
eventType$ = IF activity THEN "admin" ELSE eventType$
eventType$ = IF activity AND signin["v"] THEN "auth" ELSE eventType$
eventType$ = IF raised THEN "alarm" ELSE eventType$

# The device's own time, [Time:dd-mm@hh:mm:ss.mmm], has no year or zone, so DATETIME picks the year
# (TRUE) and reads it as UTC: set the device to UTC, or expect the times to be off by its offset. A
# relay's header time is used when there is none.
t = msg|\[Time:\s?(?P<v>\d{2}-\d{2}@\d{2}:\d{2}:\d{2}(?:\.\d+)?)\]
when = IF hdr["bsd"] THEN DATETIME(hdr["bsd"], "Jan _2 15:04:05", TRUE) ELSE eventTimeUsec$
when = IF t["v"] THEN DATETIME(t["v"], "02-01@15:04:05.999", TRUE) ELSE when
eventTimeUsec$ = when
