# Source: Barracuda Email Security Gateway (appliance) mail syslog, to the data collector.
# Child of BARRACUDA_EMAIL_SYSLOG_PARENT: RECV, SCAN and SEND lines to Email. The appliance
# writes one line per action per recipient, joined by the Barracuda message ID.
# esg, eventType$, sensorType$ and hostId$ come from the parent.

svc = esg["svc"]

# RECV: <sender> <recipient> <action> <reason> <reason extra>
recv = info|^(?P<from>\S+)\s+(?P<to>\S+)\s+(?P<action>\d+)\s+(?P<reason>\d+)\s*(?P<extra>.*)$
# SCAN: <ENC|-> <sender> <recipient> <score> <action> <reason> <reason extra> [ENC|-] SZ:<bytes> SUBJ:<subject>
scan = info|^(?P<enc>\S+)\s+(?P<from>\S+)\s+(?P<to>\S+)\s+(?P<score>\S+)\s+(?P<action>\d+)\s+(?P<reason>\d+)\s+(?P<extra>.*?)(?:\s+(?:ENC|-))?\s+SZ:(?P<size>\d+)\s+SUBJ:(?P<subj>.*)$
# SEND: <ENC|-> <action> <queue id> <response from the next server>
send = info|^(?P<enc>\S+)\s+(?P<action>\d+)\s+(?P<queue>\S+)\s+(?P<resp>.*)$

sender = IF svc == "SCAN" THEN scan["from"] ELSE recv["from"]
rcpt = IF svc == "SCAN" THEN scan["to"] ELSE recv["to"]
action = recv["action"]
action = IF svc == "SCAN" THEN scan["action"] ELSE action
action = IF svc == "SEND" THEN send["action"] ELSE action
reason = IF svc == "SCAN" THEN scan["reason"] ELSE recv["reason"]
extra = IF svc == "SCAN" THEN scan["extra"] ELSE recv["extra"]

# "-" marks an empty sender, recipient, score or reason extra.
fromEmailAddress$ = IF sender != "" AND sender != "-" THEN sender ELSE NULL
toEmailAddress$ = IF rcpt != "" AND rcpt != "-" THEN rcpt ELSE NULL
subject$ = IF scan["subj"] != "" THEN scan["subj"] ELSE NULL
messageSize$ = IF scan["size"] != "" THEN INT(scan["size"], 10) ELSE NULL
# SEND lines carry 127.0.0.1 as the client address.
senderIp$ = IF svc != "SEND" THEN esg["ip"] ELSE NULL

direction$ = NULL
direction$ = IF CONTAINS(esg["proc"], "inbound/") THEN "INBOUND" ELSE direction$
direction$ = IF esg["proc"] == "outbound/smtp" THEN "OUTBOUND" ELSE direction$

# Start is the epoch second the sender began the message. SEND lines have 0 there, so they use
# the header time, which has no zone (read as UTC) and, before 5.1.3.007, no year.
start = esg["start"]
eventTimeUsec$ = IF start != "" AND start != "0" THEN INT(start, 10) * 1000000 ELSE eventTimeUsec$
eventTimeUsec$ = IF (start == "" OR start == "0") AND esg["year"] != "" THEN DATETIME(esg["time"] + " " + esg["year"], "Jan _2 15:04:05 2006") ELSE eventTimeUsec$
eventTimeUsec$ = IF (start == "" OR start == "0") AND esg["year"] == "" THEN DATETIME(esg["time"], "Jan _2 15:04:05", TRUE) ELSE eventTimeUsec$

# Action and reason codes from Barracuda's syslog page. SEND lines use their own action codes.
actions = JSON(`{"0":"Allowed","1":"Aborted","2":"Blocked","3":"Quarantined","4":"Tagged","5":"Deferred","6":"Per-User Quarantined","7":"Allow Listed","8":"Encrypted","9":"Redirected","10":"Attachments Stubbed"}`)

Adult","70":"Intent - Political","71":"Multi-Level Intent","72":"Attachment Limit Exceeded","73":"System Busy","74":"BRTS Intent","75":"Per Domain Recipient","76":"Per Domain Sender","77":"Per Domain Client IP","78":"Sender Spoofed","79":"Attachment Content","80":"Outlook Add-in","82":"Barracuda IP/Domain Reputation","83":"Authentication Failure","85":"Attachment Size","86":"Virus detected by Extended Malware Protection","87":"Extended Malware Protection engine is busy","88":"A message was categorized for Email Category","89":"Macro Blocked"}`)
threatTypes = JSON(`{"1":"virus","86":"virus","2":"attachment","72":"attachment","79":"attachment","85":"attachment","89":"attachment","31":"spam","60":"spam","3":"reputation","62":"reputation","82":"reputation","39":"intent","65":"intent","69":"intent","70":"intent","71":"intent","74":"intent","40":"spoofing","63":"spoofing","78":"spoofing","83":"spoofing"}`)
actionName = IF svc == "SEND" THEN sendActions["$." + action] ELSE actions["$." + action]
reasonName = IF reason != "" AND reason != "0" THEN reasons["$." + reason] ELSE NULL
reasonText = IF STRING(reasonName) != "" THEN STRING(reasonName) ELSE "Reason " + reason

# Allowed, tagged and other passed-through mail is DELIVERED: the gateway let it through.
# SEND lines carry the delivery result itself.
status$ = NULL
status$ = IF svc != "SEND" AND (action == "0" OR action == "4" OR action == "7" OR action == "8" OR action == "9" OR action == "10") THEN "DELIVERED" ELSE status$
status$ = IF svc != "SEND" AND (action == "1" OR action == "2") THEN "BLOCKED" ELSE status$
status$ = IF svc != "SEND" AND (action == "3" OR action == "6") THEN "QUARANTINED" ELSE status$
status$ = IF svc == "SEND" AND action == "1" THEN "DELIVERED" ELSE status$
quarantineReason$ = IF status$ == "QUARANTINED" THEN reasonText ELSE NULL

# A blocked or quarantined message gets one threat: the reason, its kind, and the reason extra
# (the virus, filter or list that matched).
ttype = IF reason != "" THEN STRING(threatTypes["$." + reason]) ELSE ""
t_reason.name$ = reasonText
t_reason.type$ = IF ttype != "" THEN ttype ELSE "policy"
t_reason.classification$ = IF extra != "" AND extra != "-" THEN extra ELSE NULL
found = []
found = IF (status$ == "BLOCKED" OR status$ == "QUARANTINED") AND reason != "" AND reason != "0" THEN found + [t_reason] ELSE found
threats$ = found

eventMetadata$.record$ = [m1, m3, m4]

eventMetadata$.record$ = IF enc != "" THEN eventMetadata$.record$ + [m8] ELSE eventMetadata$.record$
m9.key$ = "queue_id"
m9.value$ = send["queue"]
m10.key$ = "delivery_response"
m10.value$ = send["resp"]
eventMetadata$.record$ = IF svc == "SEND" THEN eventMetadata$.record$ + [m9, m10] ELSE eventMetadata$.record$
