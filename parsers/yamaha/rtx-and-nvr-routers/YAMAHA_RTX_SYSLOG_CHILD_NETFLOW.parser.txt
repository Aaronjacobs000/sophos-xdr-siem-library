!SAMPLE=<13>Oct 14 09:20:14 rtx01 PP[01] Rejected at IN(2000) filter: TCP 203.0.113.50:3186 > 192.0.2.5:445
!CONFIRMWITH=EXPRESSION
!CONFIRMSTRING=eventType$=="filter" OR eventType$=="connection"
!SCHEMA=scwx.netflow
!TRIMALLOFF

# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Yamaha RTX and NVR routers, syslog to the data collector (syslog notice on).
# Child of YAMAHA_RTX_SYSLOG_PARENT: IP filter and NAT logs, and dynamic filter connection
# summaries, to Netflow.
#   <interface> Passed|Rejected|Restricted at IN|OUT|NAT(<filter>) filter: <protocol> <src>[:<port>] > <dst>[:<port>]
#   [INSPECT] <interface>[in|out][<filter>] <protocol> <a>:<port> >|< <b>:<port> (<start time>)
# Setting source_address and destination_address lets the IP Watchlist detector use these events.
# flt, insp, tz, eventType$, sensorType$, hostId$ and eventTimeUsec$ come from the parent.

isConn = eventType$ == "connection"

# A summary's ends are <ip>:<port>; the arrow says which end opened the connection.
aText = STRING(insp["a"])
bText = STRING(insp["b"])
am = aText|^(?P<ip>\d{1,3}(?:\.\d{1,3}){3})(?::(?P<port>\d+))?$
bm = bText|^(?P<ip>\d{1,3}(?:\.\d{1,3}){3})(?::(?P<port>\d+))?$
aIp = STRING(am["ip"])
bIp = STRING(bm["ip"])
aPort = STRING(am["port"])
bPort = STRING(bm["port"])
# a > b: a opened it. a < b: b opened it. A filter line is always source > destination.
fwd = insp["arrow"] == ">"
src = STRING(flt["src"])
dst = STRING(flt["dst"])
sport = STRING(flt["sport"])
dport = STRING(flt["dport"])
src = IF isConn AND fwd THEN aIp ELSE src
src = IF isConn AND NOT fwd THEN bIp ELSE src
dst = IF isConn AND fwd THEN bIp ELSE dst
dst = IF isConn AND NOT fwd THEN aIp ELSE dst
sport = IF isConn AND fwd THEN aPort ELSE sport
sport = IF isConn AND NOT fwd THEN bPort ELSE sport
dport = IF isConn AND fwd THEN bPort ELSE dport
dport = IF isConn AND NOT fwd THEN aPort ELSE dport
sourceAddress$ = IF src != "" THEN src ELSE NULL
destinationAddress$ = IF dst != "" THEN dst ELSE NULL
sourcePort$ = IF sport != "" THEN INT(sport, 10) ELSE NULL
destinationPort$ = IF dport != "" THEN INT(dport, 10) ELSE NULL

# protocol takes the IANA number; the router logs a name.
proto = UPPERCASE(IF isConn THEN STRING(insp["proto"]) ELSE STRING(flt["proto"]))
protocol$ = NULL
protocol$ = IF proto == "TCP" THEN 6 ELSE protocol$
protocol$ = IF proto == "UDP" THEN 17 ELSE protocol$
protocol$ = IF proto == "ICMP" THEN 1 ELSE protocol$
protocol$ = IF proto == "ICMP6" OR proto == "ICMPV6" THEN 58 ELSE protocol$
protocol$ = IF proto == "GRE" THEN 47 ELSE protocol$
protocol$ = IF proto == "ESP" THEN 50 ELSE protocol$
protocol$ = IF proto == "AH" THEN 51 ELSE protocol$

# Passed lets the packet through; Rejected drops it, and so does Restricted (a restrict filter logs
# only what it drops). A connection summary is a connection the dynamic filter let through.
act = STRING(flt["act"])
flowAction$ = IF act == "Passed" OR isConn THEN "FW_ALLOWED" ELSE "FW_BLOCKED"

# The summary's start time is the router's local time, like the header's.
start = STRING(insp["start"])
startTimestampUsec$ = IF start != "" AND tz == "" THEN DATETIME(start, "2006/01/02 15:04:05") ELSE NULL
startTimestampUsec$ = IF start != "" AND tz != "" THEN DATETIME(start + " " + tz, "2006/01/02 15:04:05 -0700") ELSE startTimestampUsec$

# Fields with no Netflow home go to event_metadata.
iface = IF isConn THEN STRING(insp["if"]) ELSE STRING(flt["if"])
where = IF isConn THEN "dynamic " + STRING(insp["dir"]) ELSE STRING(flt["at"])
num = IF isConn THEN STRING(insp["num"]) ELSE STRING(flt["num"])
rest = REPLACE_REGEX(STRING(flt["rest"]), `^:\s*`, "")
m1.key$ = "log_type"
m1.value$ = IF isConn THEN "dynamic filter connection" ELSE "IP filter"
m2.key$ = "filter"
m2.value$ = where
m3.key$ = "filter_number"
m3.value$ = num
# Assign the list first: reading eventMetadata$.record$ before it is set fails the event.
eventMetadata$.record$ = [m1, m2, m3]
m4.key$ = "interface"
m4.value$ = iface
eventMetadata$.record$ = IF iface != "" THEN eventMetadata$.record$ + [m4] ELSE eventMetadata$.record$
m5.key$ = "filter_action"
m5.value$ = act
eventMetadata$.record$ = IF act != "" THEN eventMetadata$.record$ + [m5] ELSE eventMetadata$.record$
# The ICMP type of an ICMP line, or the query of a DNS line.
m6.key$ = "detail"
m6.value$ = rest
eventMetadata$.record$ = IF rest != "" THEN eventMetadata$.record$ + [m6] ELSE eventMetadata$.record$
