# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Trellix Intrusion Prevention System (formerly McAfee Network Security Platform) syslog
# from the Manager to the data collector. Messages are built from $IV_ variables, so this entry's
# README gives a template for IPS alerts and one for user activity (audit), each a list of
# KEY=value pairs joined by |:
#   TRELLIX_IPS=alert|ALERT_ID=...|ATTACK_NAME=...|SOURCE_IP=...|...
# It also reads the Manager's default alert message:
#   <sensor> detected <direction> attack <name> (severity = <severity>). <src>:<port> -> <dst>:<port> (result = <result>)
# From 11.1 Update 5 the Manager adds a [Mmm dd, yyyy hh:mm:ss] time to every message; the
# parser takes it off either end and uses it as the event time.
# Parent: hands alerts to the NIDS child and user activity to the ManagementEvent child.

# The Manager puts its own text right after the syslog time, so the parser looks for the
# template or the default message anywhere after it, past any tag a relay adds.
m = originalData$|^(?:<\d{1,3}>)?(?:(?P<time>[A-Z][a-z]{2}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+)?(?P<rest>.*)$
rest = m["rest"]
tpl = rest|TRELLIX_IPS=(?P<kind>alert|audit)\|(?P<body>.*?)(?:\s*\[[A-Z][a-z]{2} \d{1,2}, \d{4} \d{2}:\d{2}:\d{2}\])?\s*$
kv = IF tpl["kind"] != "" THEN SPLIT_NAME_VALUES(tpl["body"], "|", "=", "\\") ELSE NULL
d = rest|(?:^|\s)(?P<sensor>[^\s\[\]]+) detected (?P<dir>\S+) attack (?P<name>.+) \(severity = (?P<sev>[^)]*)\)\.\s+(?P<src>\S+?):(?P<sport>\d+|N/A) -> (?P<dst>\S+?):(?P<dport>\d+|N/A) \(result = (?P<result>[^)]*)\)
stamp = rest|\[(?P<v>[A-Z][a-z]{2} \d{1,2}, \d{4} \d{2}:\d{2}:\d{2})\]

# Variables and $ fields set here are visible to both children.
sensorType$ = "Trellix IPS"
# The Manager's times have no zone and are read as UTC. The bracketed time has the year.
eventTimeUsec$ = IF m["time"] != "" THEN DATETIME(m["time"], "Jan _2 15:04:05", TRUE) ELSE eventTimeUsec$
eventTimeUsec$ = IF stamp["v"] != "" THEN DATETIME(stamp["v"], "Jan _2, 2006 15:04:05") ELSE eventTimeUsec$

eventType$ = "other"
eventType$ = IF d["name"] != "" OR tpl["kind"] == "alert" THEN "alert" ELSE eventType$
eventType$ = IF tpl["kind"] == "audit" THEN "audit" ELSE eventType$
