# Parser Studio 0.1.0: https://github.com/Aaronjacobs000/sophos-xdr-siem-library
# Emulator result, not run in Sophos NG-SIEM. Test with Parse Message and Run Test before you turn it on.
# Source: Docker container logs from the syslog logging driver (log-driver syslog), sent to the
# data collector. The tag is the container's short ID by default ({{.ID}}), or a template that
# ends with it, such as {{.Name}}/{{.ID}}:
#   default:                 <PRI><Mon> <day> <hh:mm:ss> <tag>[<dockerd pid>]: <line>
#   rfc3164:                 <PRI><Mon> <day> <hh:mm:ss> <docker host> <tag>[<dockerd pid>]: <line>
#   rfc5424 and rfc5424micro: <PRI>1 <time with offset> <docker host> <tag> <dockerd pid> <tag> - <line>
# Standalone parser for the generic schema: a container's output is the application's own text,
# with no fields the parser can know. It sets the sensor type, the Docker host, the container
# (the tag) as the sensor ID, and the time.

s = originalData$|^(?:<\d{1,3}>)?(?P<bsd>[A-Z][a-z]{2}\s+\d{1,2}\s\d{2}:\d{2}:\d{2})\s(?:(?P<host>\S+)\s)?(?P<tag>(?:\S*/)?[0-9a-f]{12})\[\d+\]:\s
r = originalData$|^(?:<\d{1,3}>)?1\s(?P<iso>\d{4}-\d{2}-\d{2}T\S+)\s(?P<host>\S+)\s(?P<tag>(?:\S*/)?[0-9a-f]{12})\s\d+\s
is5424 = STRING(r["iso"]) != ""

sensorType$ = "Docker"
host = IF is5424 THEN STRING(r["host"]) ELSE STRING(s["host"])
hostId$ = IF host != "" THEN host ELSE NULL
sensorId$ = IF is5424 THEN r["tag"] ELSE s["tag"]
# rfc5424 times carry the offset. The default and rfc3164 times are dockerd's local time with no
# year or zone, read as UTC in the current year, so use rfc5424 unless the Docker host runs on UTC.
eventTimeUsec$ = IF is5424 THEN DATETIME(r["iso"], "2006-01-02T15:04:05.999999999Z07:00") ELSE DATETIME(s["bsd"], "Jan _2 15:04:05", TRUE)
