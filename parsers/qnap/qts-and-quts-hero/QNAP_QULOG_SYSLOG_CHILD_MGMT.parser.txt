# Source: QNAP QuLog Center event log lines sent to a syslog server.
# Child of QNAP_QULOG_SYSLOG_PARENT: event log lines that name a user, such as created users,
# changed passwords and new shared folders, to Management Event, with the user, the address they
# worked from, the app and category, and the event text as the operation. Event log lines from
# the system (Users: --- or System) stay generic.
# user, src, comp, nas, ev, eventType$, sensorType$, hostId$ and eventTimeUsec$ come from the parent.

channel$ = "QuLog Center event log"
username$ = user
clientHostname$ = IF src != "" THEN src ELSE NULL
targetHostname$ = nas
app = ev["app"]
cat = ev["cat"]
operationName$ = IF cat != "" THEN app + ": " + cat ELSE app
operation$ = ev["content"]
